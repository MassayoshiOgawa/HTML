function weightedRandom(types) {
  const entries = Object.entries(types);
  const totalWeight = entries.reduce((sum, [, t]) => sum + t.weight, 0);
  let r = Math.random() * totalWeight;
  for (const [key, t] of entries) {
    r -= t.weight;
    if (r <= 0) return key;
  }
  return entries[entries.length - 1][0];
}

// ============================================================
// MAP PARSER
// ============================================================
function parseMap(text) {
  const lines = text.split('\n').filter(l => l.length > 0);
  if (lines.length === 0) return null;
  let cleanLines = [];
  let cols = 0;
  for (const line of lines) {
    const clean = line.replace(/ /g, '');
    if (clean.length > 0) {
      cleanLines.push(clean);
      if (clean.length > cols) cols = clean.length;
    }
  }
  if (cleanLines.length === 0) return null;
  // Pad rows to same width
  for (let i = 0; i < cleanLines.length; i++) {
    while (cleanLines[i].length < cols) cleanLines[i] += 'G';
  }
   const validChars = new Set(['O','C','G','d','A','P','M','F','U','W','S','T','m']);
  const tiles = [];
  const doors = [];
   const artifacts = [];
   const waypoints = [];
   const randomSpawns = [];
   let altarPos = null;
   let playerSpawn = null;
   let monsterSpawn = null;
   for (let y = 0; y < cleanLines.length; y++) {
     const row = [];
     for (let x = 0; x < cleanLines[y].length; x++) {
       const ch = cleanLines[y][x];
       if (!validChars.has(ch)) {
         console.warn('Invalid char:', ch, 'at', x, y);
         row.push('C');
         continue;
       }
       row.push(ch);
       if (ch === 'd') doors.push({ x, y });
       if (ch === 'F') artifacts.push({ x, y });
       if (ch === 'U') waypoints.push({ x, y });
       if (ch === 'A') altarPos = { x, y };
       if (ch === 'P') playerSpawn = { x, y };
       if (ch === 'M') monsterSpawn = { x, y };
       if (ch === 'm') randomSpawns.push({ x, y });
     }
     tiles.push(row);
   }
   return {
     tiles, doors, artifacts, waypoints, randomSpawns,
     altarPos, playerSpawn, monsterSpawn,
     width: cols, height: cleanLines.length
   };
}

// ============================================================
// PARTICLE SYSTEM
// ============================================================
class Particle {
  constructor(x, y, vx, vy, life, color) {
    this.x = x; this.y = y;
    this.vx = vx; this.vy = vy;
    this.life = life; this.maxLife = life;
    this.color = color;
  }
  update(dt) {
    this.x += this.vx * dt;
    this.y += this.vy * dt;
    this.life -= dt;
  }
  get alive() { return this.life > 0; }
  render(ctx, cx, cy) {
    const alpha = this.life / this.maxLife;
    ctx.globalAlpha = alpha;
    ctx.fillStyle = this.color;
    ctx.fillRect(this.x - cx, this.y - cy, 2, 2);
    ctx.globalAlpha = 1;
  }
}

// ============================================================
// DOOR ENTITY
// ============================================================
class DoorEntity {
  constructor(x, y) {
    this.tileX = x; this.tileY = y;
    this.open = false;
    this.animating = false;
    this.animTimer = 0;
    this.animDuration = 0.4;
    this.debounceTimer = 0;
    this.animProgress = 0;
    this.cooldown = false;
    this.groupMembers = null;
  }
  hasGroup() { return this.groupMembers !== null && this.groupMembers.length > 0; }
  update(dt) {
    if (this.animating) {
      this.animTimer += dt;
      if (this.open) this.animProgress = Math.min(1, this.animTimer / this.animDuration);
      else this.animProgress = Math.max(0, 1 - this.animTimer / this.animDuration);
      if (this.animTimer >= this.animDuration) {
        this.animating = false;
        this.debounceTimer = 0.3;
        this.animProgress = this.open ? 1 : 0;
      }
    }
    if (this.debounceTimer > 0) {
      this.debounceTimer -= dt;
      if (this.debounceTimer <= 0) this.debounceTimer = 0;
    }
  }
  toggle(skipPropagation) {
    if (this.animating || this.debounceTimer > 0) return;
    if (!skipPropagation && this.groupMembers) {
      for (let i = 0; i < this.groupMembers.length; i++) {
        const member = this.groupMembers[i];
        if (member !== this) member.toggle(true);
      }
    }
    this.open = !this.open;
    this.animating = true;
    this.animTimer = 0;
  }
  forceOpen() {
    if (!this.open && !this.animating) {
      if (this.groupMembers) {
        for (let i = 0; i < this.groupMembers.length; i++) {
          const member = this.groupMembers[i];
          if (member !== this && !member.open && !member.animating) member.toggle(true);
        }
      }
      this.toggle(true);
    }
  }
  get solid() { return !this.open && !this.animating; }
}

class ArtifactEntity {
  constructor(x, y) {
    this.tileX = x; this.tileY = y;
    this.collected = false;
    this.glowPhase = Math.random() * Math.PI * 2;
  }
  update(dt) { this.glowPhase += dt * 3; }
  render(ctx, tx, ty, ts, camX, camY) {
    if (this.collected) return;
    const cx = tx * ts + ts / 2 - camX;
    const cy = ty * ts + ts / 2 - camY;
    const glow = 0.5 + 0.5 * Math.sin(this.glowPhase);
    ctx.fillStyle = '#00cccc';
    ctx.globalAlpha = 0.3 + glow * 0.7;
    ctx.beginPath();
    ctx.arc(cx, cy, ts * 0.3, 0, Math.PI * 2);
    ctx.fill();
    ctx.globalAlpha = 1;
    ctx.fillStyle = '#00cccc';
    ctx.beginPath();
    ctx.arc(cx, cy, ts * 0.12, 0, Math.PI * 2);
    ctx.fill();
  }
}

// ============================================================
// PLAYER
// ============================================================
class Player {
  constructor(tileX, tileY, playerIndex, name, playerClass) {
    this.playerIndex = playerIndex || 0;
    this.name = name || ('P' + (this.playerIndex + 1));
    this.tileX = tileX;
    this.tileY = tileY;
    this.px = tileX * TILE_SIZE + TILE_SIZE / 2;
    this.py = tileY * TILE_SIZE + TILE_SIZE / 2;
    this.speed = 140;
    this.baseSpeed = 140;
    this.hp = 100;
    this.maxHp = 100;
    this.alive = true;
    this.interacting = false;
    this.interactTimer = 0;
    this.interactTarget = null;
    this.interactType = null;
    this.invulnTimer = 0;
    this.playerClass = playerClass || 'none';
    this.abilityCooldown = 0;
    this.abilityActive = false;
    this.abilityTimer = 0;
    this.keybinds = { ...DEFAULT_KEYBINDS[this.playerIndex] };
    this.inventory = [null, null, null, null, null];
    this.selectedSlot = 0;
    this.useTimer = 0;
    this.using = false;
  }

  update(dt, game) {
    if (this.invulnTimer > 0) this.invulnTimer -= dt;
    if (!this.alive) return;
    if (this.abilityCooldown > 0) this.abilityCooldown -= dt;
    if (this.abilityActive) {
      this.abilityTimer -= dt;
      if (this.abilityTimer <= 0) {
        this.abilityActive = false;
        if (this.playerClass === 'scout') this.speed = this.baseSpeed;
      }
    }
    let dx = 0, dy = 0;
    const kb = this.keybinds;
    if (keys[kb.up]) dy -= 1;
    if (keys[kb.down]) dy += 1;
    if (keys[kb.left]) dx -= 1;
    if (keys[kb.right]) dx += 1;
    if (dx !== 0 && dy !== 0) { dx *= 0.707; dy *= 0.707; }
    const spd = this.speed;
    const nx = this.px + dx * spd * dt;
    const ny = this.py + dy * spd * dt;
    if (this.canMoveTo(nx, this.py, game)) this.px = nx;
    if (this.canMoveTo(this.px, ny, game)) this.py = ny;
    this.tileX = Math.floor(this.px / TILE_SIZE);
    this.tileY = Math.floor(this.py / TILE_SIZE);
    this.checkInteraction(game);
    if (keysJustPressed[kb.interact] && this.interactTarget && this.interactType === 'door') {
      this.interactTarget.toggle();
      keysJustPressed[kb.interact] = false;
    }
    if (this.interacting && this.interactType !== 'door') {
      this.interactTimer += dt;
      game.updateInteraction(this);
    }
    if (!keys[kb.interact]) { this.interacting = false; this.interactTimer = 0; }
    // Hotbar: cycle slot on tap, hold to use
    if (keysJustPressed[kb.use]) {
      // Tap R: cycle slot (skip empty)
      keysJustPressed[kb.use] = false;
      let next = this.selectedSlot;
      let tries = 0;
      do {
        next = (next + 1) % 5;
        tries++;
      } while (tries < 5 && !this.inventory[next]);
      this.selectedSlot = next;
      this.useTimer = 0;
      this.using = false;
    }
    // Hold R to use item
    if (keys[kb.use] && this.inventory[this.selectedSlot] && !this.using) {
      this.useTimer += dt;
      if (this.useTimer >= 1) {
        game.useItem(this);
        this.useTimer = 0;
      }
    }
    if (!keys[kb.use]) { this.useTimer = 0; this.using = false; }
  }

  hasSpace() { return this.inventory.some(s => !s); }
  addToInventory(item) {
    for (let i = 0; i < 5; i++) {
      if (!this.inventory[i]) { this.inventory[i] = item; return true; }
    }
    return false;
  }
  getEquippedItem() { return this.inventory[this.selectedSlot]; }
  removeEquipped() { this.inventory[this.selectedSlot] = null; }

  canMoveTo(px, py, game) {
    const margin = 4;
    const checks = [
      { x: px - margin, y: py - margin },
      { x: px + margin, y: py - margin },
      { x: px - margin, y: py + margin },
      { x: px + margin, y: py + margin }
    ];
    for (const c of checks) {
      const tx = Math.floor(c.x / TILE_SIZE);
      const ty = Math.floor(c.y / TILE_SIZE);
      if (tx < 0 || ty < 0 || tx >= game.mapWidth || ty >= game.mapHeight) return false;
      const tile = game.tiles[ty][tx];
      if (tile === 'O' || tile === 'W') return false;
      if (tile === 'd') {
        const door = game.doorEntities.find(d => d.tileX === tx && d.tileY === ty);
        if (door && door.solid) return false;
      }
    }
    return true;
  }

  checkInteraction(game) {
    const range = 1.5;
    this.interactTarget = null;
    let bestDist = range * TILE_SIZE;
    for (const door of game.doorEntities) {
      const dx = door.tileX + 0.5 - this.px / TILE_SIZE;
      const dy = door.tileY + 0.5 - this.py / TILE_SIZE;
      const dist = Math.sqrt(dx * dx + dy * dy) * TILE_SIZE;
      if (dist < bestDist) {
        bestDist = dist;
        this.interactTarget = door;
        this.interactType = 'door';
      }
    }
    if (game.altarPos && game.activeMonsters.every(m => !m.alive)) {
      const dx = game.altarPos.x + 0.5 - this.px / TILE_SIZE;
      const dy = game.altarPos.y + 0.5 - this.py / TILE_SIZE;
      const dist = Math.sqrt(dx * dx + dy * dy) * TILE_SIZE;
      if (dist < bestDist) {
        bestDist = dist;
        this.interactTarget = 'altar';
        this.interactType = 'altar';
      }
    }
    for (const art of game.artifactEntities) {
      if (art.collected) continue;
      const dx = art.tileX + 0.5 - this.px / TILE_SIZE;
      const dy = art.tileY + 0.5 - this.py / TILE_SIZE;
      const dist = Math.sqrt(dx * dx + dy * dy) * TILE_SIZE;
      if (dist < bestDist) {
        bestDist = dist;
        this.interactTarget = art;
        this.interactType = 'artifact';
      }
    }
    // Check shops (only when monster dead)
    if (game.activeMonsters.every(m => !m.alive)) {
      for (const shop of game.shopItems) {
        if (shop.sold) continue;
        const dx = shop.x + 0.5 - this.px / TILE_SIZE;
        const dy = shop.y + 0.5 - this.py / TILE_SIZE;
        const dist = Math.sqrt(dx * dx + dy * dy) * TILE_SIZE;
        if (dist < bestDist) {
          bestDist = dist;
          this.interactTarget = shop;
          this.interactType = 'shop';
        }
      }
    }
    // Check trash bins (only when monster dead)
    if (game.activeMonsters.every(m => !m.alive)) {
      for (let y = 0; y < game.mapHeight; y++) {
        for (let x = 0; x < game.mapWidth; x++) {
          if (game.tiles[y][x] !== 'T') continue;
          const dx = x + 0.5 - this.px / TILE_SIZE;
          const dy = y + 0.5 - this.py / TILE_SIZE;
          const dist = Math.sqrt(dx * dx + dy * dy) * TILE_SIZE;
          if (dist < bestDist) {
            bestDist = dist;
            this.interactTarget = { x, y };
            this.interactType = 'trash';
          }
        }
      }
    }
  }

  takeDamage(dmg, game) {
    if (this.invulnTimer > 0) return;
    this.hp -= dmg;
    this.invulnTimer = 0.8;
    for (let i = 0; i < 8; i++) {
      game.particles.push(new Particle(
        this.px, this.py,
        (Math.random() - 0.5) * 80, (Math.random() - 0.5) * 80,
        0.4, '#ff4444'
      ));
    }
    if (this.hp <= 0) {
      this.hp = 0; this.alive = false;
      game.showMessage(this.name + ' died!');
    }
  }

  render(ctx, game) {
    if (!this.alive) return;
    if (this.invulnTimer > 0 && Math.floor(this.invulnTimer * 10) % 2 === 0) return;
    if (!game.debugMode && !game.isPointVisible(this.px, this.py)) return;
    const sx = this.px - game.camX;
    const sy = this.py - game.camY;
    const playerColors = ['#4488ff', '#ff8844', '#44ff44', '#ff44ff'];
    const playerStrokes = ['#2266dd', '#dd6622', '#22dd22', '#dd22dd'];
    const ci = this.playerIndex % playerColors.length;
    ctx.fillStyle = playerColors[ci];
    ctx.beginPath();
    ctx.arc(sx, sy, TILE_SIZE * 0.35, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = playerStrokes[ci];
    ctx.lineWidth = 1.5;
    ctx.stroke();
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(sx - 3, sy - 2, 3, 0, Math.PI * 2);
    ctx.arc(sx + 3, sy - 2, 3, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#111';
    ctx.beginPath();
    ctx.arc(sx - 3, sy - 2, 1.5, 0, Math.PI * 2);
    ctx.arc(sx + 3, sy - 2, 1.5, 0, Math.PI * 2);
    ctx.fill();
    // Name label
    ctx.font = '10px "Courier New", monospace';
    ctx.fillStyle = '#fff';
    ctx.textAlign = 'center';
    ctx.fillText(this.name, sx, sy - TILE_SIZE * 0.55);
    ctx.textAlign = 'start';
  }
}

// ============================================================
// MONSTER
// ============================================================
class MonsterEntity {
  constructor(tileX, tileY, game, monsterType) {
    this.tileX = tileX;
    this.tileY = tileY;
    this.px = tileX * TILE_SIZE + TILE_SIZE / 2;
    this.py = tileY * TILE_SIZE + TILE_SIZE / 2;
    this.speed = 70;
    this.alive = false;
    this.state = 'patrol';
    this.currentWaypoint = 0;
    this.waypointOrder = [];
    this.facingAngle = 0;
    this.chaseTimer = 0;
    this.searchTimer = 0;
    this.lastKnownPlayerPos = null;
    this.attackCooldown = 0;
    this.game = game;
    this.path = null;
    this.pathTarget = null;
    this.pathRecalc = 0;
    this.type = monsterType || 'normal';
    const mt = MonsterTypes[this.type];
    this.chaseLoseTime = mt.chaseLoseTime;
    this.searchDuration = mt.searchDuration;
    this.bodyColor = mt.color;
    this.bodyStroke = mt.strokeColor;
    this.name = mt.name;
    this.stunTimer = 0;
  }

  spawn() {
    this.alive = true;
    this.state = 'patrol';
    this.currentWaypoint = 0;
    this.generateWaypointOrder();
    if (this.waypointOrder.length > 0) this.currentWaypoint = 0;
    this.px = this.tileX * TILE_SIZE + TILE_SIZE / 2;
    this.py = this.tileY * TILE_SIZE + TILE_SIZE / 2;
    this.path = null;
    this.pathTarget = null;
  }

  die() {
    this.alive = false;
    this.state = 'patrol';
    this.path = null;
    this.pathTarget = null;
  }

  generateWaypointOrder() {
    const wp = this.game.waypoints;
    if (wp.length === 0) { this.waypointOrder = []; return; }
    const indices = [...Array(wp.length).keys()];
    for (let i = indices.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [indices[i], indices[j]] = [indices[j], indices[i]];
    }
    this.waypointOrder = indices;
    this.currentWaypoint = 0;
  }

  findNearestPlayer() {
    let nearest = null;
    let nearestDist = Infinity;
    for (const p of this.game.players) {
      if (!p.alive) continue;
      const dx = p.px - this.px;
      const dy = p.py - this.py;
      const dist = Math.sqrt(dx * dx + dy * dy);
      if (dist < nearestDist) { nearestDist = dist; nearest = p; }
    }
    return nearest;
  }

  canSeePlayer() {
    const player = this.findNearestPlayer();
    if (!player) return false;
    const dx = player.px - this.px;
    const dy = player.py - this.py;
    const dist = Math.sqrt(dx * dx + dy * dy);
    if (dist > 250) return false;
    const angle = Math.atan2(dy, dx);
    let diff = angle - this.facingAngle;
    while (diff > Math.PI) diff -= Math.PI * 2;
    while (diff < -Math.PI) diff += Math.PI * 2;
    if (Math.abs(diff) > (Math.PI / 3) * 0.85 && dist > 50) return false;
    return this.hasLineOfSight(this.px, this.py, player.px, player.py);
  }

  canDetectBehind() {
    const player = this.findNearestPlayer();
    if (!player) return false;
    const dx = player.px - this.px;
    const dy = player.py - this.py;
    const dist = Math.sqrt(dx * dx + dy * dy);
    if (dist > 45) return false;
    const angleToPlayer = Math.atan2(dy, dx);
    let diff = angleToPlayer - this.facingAngle;
    while (diff > Math.PI) diff -= Math.PI * 2;
    while (diff < -Math.PI) diff += Math.PI * 2;
    if (Math.abs(diff) > Math.PI * 0.65) return true;
    return false;
  }

  hasLineOfSight(x1, y1, x2, y2) {
    const tiles = this.game.tiles;
    const tx1 = Math.floor(x1 / TILE_SIZE);
    const ty1 = Math.floor(y1 / TILE_SIZE);
    const tx2 = Math.floor(x2 / TILE_SIZE);
    const ty2 = Math.floor(y2 / TILE_SIZE);
    const dx = Math.abs(tx2 - tx1);
    const dy = Math.abs(ty2 - ty1);
    const sx = tx1 < tx2 ? 1 : -1;
    const sy = ty1 < ty2 ? 1 : -1;
    let err = dx - dy;
    let cx = tx1, cy = ty1;
    const steps = dx + dy;
    for (let i = 0; i < steps; i++) {
      if (cx === tx2 && cy === ty2) break;
      const e2 = err * 2;
      if (e2 > -dy) { err -= dy; cx += sx; }
      if (e2 < dx) { err += dx; cy += sy; }
      if (cx >= 0 && cx < this.game.mapWidth && cy >= 0 && cy < this.game.mapHeight) {
        if (tiles[cy][cx] === 'O') return false;
        if (tiles[cy][cx] === 'W') return false;
        if (tiles[cy][cx] === 'd') {
          const door = this.game.doorEntities.find(d => d.tileX === cx && d.tileY === cy);
          if (!door || door.solid) return false;
        }
      }
    }
    return true;
  }

  canMoveTo(px, py) {
    const margin = 3;
    const checks = [
      { x: px - margin, y: py - margin },
      { x: px + margin, y: py - margin },
      { x: px - margin, y: py + margin },
      { x: px + margin, y: py + margin }
    ];
    for (const c of checks) {
      const tx = Math.floor(c.x / TILE_SIZE);
      const ty = Math.floor(c.y / TILE_SIZE);
      if (tx < 0 || ty < 0 || tx >= this.game.mapWidth || ty >= this.game.mapHeight) return false;
      const tile = this.game.tiles[ty][tx];
      if (tile === 'O' || tile === 'W') return false;
      if (tile === 'd') {
        const door = this.game.doorEntities.find(d => d.tileX === tx && d.tileY === ty);
        if (door && door.solid) return false;
      }
    }
    return true;
  }

  update(dt) {
    if (!this.alive) return;
    if (this.stunTimer > 0) {
      this.stunTimer -= dt;
      if (this.stunTimer <= 0) this.stunTimer = 0;
      return;
    }
    if (this.attackCooldown > 0) this.attackCooldown -= dt;
    this.checkNearbyDoors(dt);
    switch (this.state) {
      case 'patrol': this.updatePatrol(dt); break;
      case 'chase': this.updateChase(dt); break;
      case 'search': this.updateSearch(dt); break;
    }
    this.tileX = Math.floor(this.px / TILE_SIZE);
    this.tileY = Math.floor(this.py / TILE_SIZE);
    // Attack nearest alive player
    if (this.alive) {
      const target = this.findNearestPlayer();
      if (target) {
        const dx = target.px - this.px;
        const dy = target.py - this.py;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < TILE_SIZE * 0.8 && this.attackCooldown <= 0) {
          target.takeDamage(20, this.game);
          this.attackCooldown = 1.0;
        }
      }
    }
  }

  checkNearbyDoors(dt) {
    const tx = Math.floor(this.px / TILE_SIZE);
    const ty = Math.floor(this.py / TILE_SIZE);
    for (let dy = -1; dy <= 1; dy++) {
      for (let dx = -1; dx <= 1; dx++) {
        const cx = tx + dx, cy = ty + dy;
        if (cx < 0 || cy < 0 || cx >= this.game.mapWidth || cy >= this.game.mapHeight) continue;
        if (this.game.tiles[cy][cx] === 'd') {
          const door = this.game.doorEntities.find(d => d.tileX === cx && d.tileY === cy);
          if (door && !door.open) door.forceOpen();
        }
      }
    }
  }

  followPath(dt) {
    if (!this.path || this.path.length === 0) return false;
    const next = this.path[0];
    const tx = next.x * TILE_SIZE + TILE_SIZE / 2;
    const ty = next.y * TILE_SIZE + TILE_SIZE / 2;
    const dx = tx - this.px;
    const dy = ty - this.py;
    const dist = Math.sqrt(dx * dx + dy * dy);
    if (dist < TILE_SIZE * 0.25) {
      this.path.shift();
      if (this.path.length === 0) return false;
      return this.followPath(dt);
    }
    const ndx = dx / dist, ndy = dy / dist;
    this.facingAngle = Math.atan2(ndy, ndx);
    const spd = this.state === 'chase' ? this.speed * 1.4 : this.speed;
    const nx = this.px + ndx * spd * dt;
    const ny = this.py + ndy * spd * dt;
    if (this.canMoveTo(nx, this.py)) this.px = nx;
    if (this.canMoveTo(this.px, ny)) this.py = ny;
    return true;
  }

  updatePatrol(dt) {
    if (this.canDetectBehind()) {
      this.state = 'chase'; this.chaseTimer = 0; this.path = null; return;
    }
    if (this.canSeePlayer()) {
      this.state = 'chase'; this.chaseTimer = 0; this.path = null; return;
    }
    if (this.waypointOrder.length === 0) return;
    const wpIdx = this.waypointOrder[this.currentWaypoint];
    if (wpIdx >= this.game.waypoints.length) {
      this.generateWaypointOrder(); return;
    }
    const wp = this.game.waypoints[wpIdx];
    this.pathRecalc -= dt;
    if (!this.path || this.pathTarget !== wpIdx || this.pathRecalc <= 0) {
      const stx = Math.floor(this.px / TILE_SIZE);
      const sty = Math.floor(this.py / TILE_SIZE);
      const found = this.game.findPath(stx, sty, wp.x, wp.y);
      if (found) {
        this.path = found;
        this.pathTarget = wpIdx;
        this.pathRecalc = 0.5;
      } else {
        this.path = null;
        this.pathTarget = null;
      }
    }
    if (this.path && this.path.length > 0) {
      if (!this.followPath(dt)) {
        this.path = null;
        this.currentWaypoint++;
        if (this.currentWaypoint >= this.waypointOrder.length) {
          this.generateWaypointOrder();
        }
      }
    } else {
      this.currentWaypoint++;
      if (this.currentWaypoint >= this.waypointOrder.length) {
        this.generateWaypointOrder();
      }
    }
  }

  updateChase(dt) {
    const target = this.findNearestPlayer();
    if (!target) { this.state = 'patrol'; this.path = null; return; }
    if (!this.canSeePlayer() && !this.canDetectBehind()) {
      this.chaseTimer += dt;
      if (this.chaseTimer > this.chaseLoseTime) {
        this.lastKnownPlayerPos = {
          x: target.px, y: target.py
        };
        this.state = 'search';
        this.searchTimer = 0;
        this.path = null;
        return;
      }
    } else {
      this.chaseTimer = 0;
    }
    this.lastKnownPlayerPos = {
      x: target.px, y: target.py
    };
    const stx = Math.floor(this.px / TILE_SIZE);
    const sty = Math.floor(this.py / TILE_SIZE);
    const gtx = Math.floor(target.px / TILE_SIZE);
    const gty = Math.floor(target.py / TILE_SIZE);
    const found = this.game.findPath(stx, sty, gtx, gty);
    if (found && found.length > 0) {
      this.path = found;
      this.followPath(dt);
    }
  }

  updateSearch(dt) {
    if (this.canSeePlayer() || this.canDetectBehind()) {
      this.state = 'chase'; this.chaseTimer = 0; this.path = null; return;
    }
    this.searchTimer += dt;
    if (this.searchTimer > this.searchDuration || !this.lastKnownPlayerPos) {
      this.state = 'patrol'; this.path = null; return;
    }
    const stx = Math.floor(this.px / TILE_SIZE);
    const sty = Math.floor(this.py / TILE_SIZE);
    const gtx = Math.floor(this.lastKnownPlayerPos.x / TILE_SIZE);
    const gty = Math.floor(this.lastKnownPlayerPos.y / TILE_SIZE);
    if (!this.path) {
      const found = this.game.findPath(stx, sty, gtx, gty);
      if (found && found.length > 0) {
        this.path = found;
      } else {
        this.state = 'patrol'; this.path = null; return;
      }
    }
    if (!this.followPath(dt)) {
      this.state = 'patrol'; this.path = null;
    }
  }

  render(ctx, game) {
    if (!this.alive) return;
    const oracleReveal = game.players.some(p => p.playerClass === 'oracle' && p.abilityActive);
    if (!game.debugMode && !oracleReveal && !game.isPointVisible(this.px, this.py)) return;
    const rx = this.px - game.camX;
    const ry = this.py - game.camY;
    if (oracleReveal) {
      ctx.fillStyle = 'rgba(255,255,0,0.25)';
      ctx.beginPath();
      ctx.arc(rx, ry, TILE_SIZE * 0.7, 0, Math.PI * 2);
      ctx.fill();
    }
    // Body
    ctx.fillStyle = this.bodyColor;
    ctx.beginPath();
    ctx.arc(rx, ry, TILE_SIZE * 0.38, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = this.bodyStroke;
    ctx.lineWidth = 1.5;
    ctx.stroke();
    // Direction indicator
    const ex = rx + Math.cos(this.facingAngle) * TILE_SIZE * 0.5;
    const ey = ry + Math.sin(this.facingAngle) * TILE_SIZE * 0.5;
    ctx.fillStyle = '#ff0';
    ctx.beginPath();
    ctx.arc(ex, ey, 3, 0, Math.PI * 2);
    ctx.fill();
    // Hat
    ctx.fillStyle = '#ffcc00';
    ctx.fillRect(rx - TILE_SIZE * 0.15, ry - TILE_SIZE * 0.45, TILE_SIZE * 0.3, TILE_SIZE * 0.12);
    ctx.fillRect(rx - TILE_SIZE * 0.08, ry - TILE_SIZE * 0.6, TILE_SIZE * 0.16, TILE_SIZE * 0.18);
    // Eyes
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(rx - 3, ry - 2, 3, 0, Math.PI * 2);
    ctx.arc(rx + 3, ry - 2, 3, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#f00';
    ctx.beginPath();
    ctx.arc(rx - 3, ry - 2, 1.5, 0, Math.PI * 2);
    ctx.arc(rx + 3, ry - 2, 1.5, 0, Math.PI * 2);
    ctx.fill();
  }

  renderVisionCone(ctx, game) {
    if (!this.alive || !game.debugMode) return;
    const rx = this.px - game.camX;
    const ry = this.py - game.camY;
    const r = 195;
    // Front vision cone
    ctx.strokeStyle = 'rgba(255,0,0,0.3)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.arc(rx, ry, r, this.facingAngle - Math.PI / 3, this.facingAngle + Math.PI / 3);
    ctx.lineTo(rx, ry);
    ctx.closePath();
    ctx.stroke();
    // Rear detection zone (45px range, 126 deg sector behind monster)
    const rearR = 45;
    ctx.fillStyle = 'rgba(255, 165, 0, 0.15)';
    ctx.beginPath();
    ctx.arc(rx, ry, rearR, this.facingAngle + Math.PI * 0.65, this.facingAngle - Math.PI * 0.65);
    ctx.lineTo(rx, ry);
    ctx.closePath();
    ctx.fill();
    ctx.strokeStyle = 'rgba(255, 165, 0, 0.5)';
    ctx.lineWidth = 1;
    ctx.stroke();
  }
}

  static tileHash(x, y) {
    let h = (x * 374761393 + y * 668265263) ^ 0x5bf03635;
    h = ((h >> 16) ^ h) * 0x45d9f3b;
    return ((h >>> 0) % 1000) / 1000;
  }

  static progressColor(t) {
    const r = Math.min(255, Math.max(0, Math.round(t < 0.5 ? 255 : 255 * (1 - t) * 2)));
    const g = Math.min(255, Math.max(0, Math.round(t < 0.5 ? 255 * t * 2 : 255)));
    const b = 0;
    return 'rgb(' + r + ',' + g + ',' + b + ')';
  }

  resolveBaseTile(x, y) {
    const tile = this.tiles[y][x];
    if (tile !== 'F' && tile !== 'U' && tile !== 'm') return tile;
    let floorCount = 0;
    let grassCount = 0;
    for (let dy = -1; dy <= 1; dy++) {
      for (let dx = -1; dx <= 1; dx++) {
        if (dx === 0 && dy === 0) continue;
        const nx = x + dx, ny = y + dy;
        if (nx < 0 || nx >= this.mapWidth || ny < 0 || ny >= this.mapHeight) continue;
        const neighborTile = this.tiles[ny][nx];
        if (neighborTile === 'C') floorCount++;
        else if (neighborTile === 'G') grassCount++;
        else if (neighborTile === 'd') floorCount++; // door = indoor
      }
    }
    return floorCount >= grassCount ? 'C' : 'G';
  }

  getEffectiveTile(x, y) {
    if (this.baseTileResolved[y] && this.baseTileResolved[y][x] !== undefined) {
      return this.baseTileResolved[y][x];
    }
    return this.tiles[y][x];
  }

  loadMap(mapData) {
    if (!mapData) return;
    this.tiles = mapData.tiles;
    this.mapWidth = mapData.width;
    this.mapHeight = mapData.height;
    this.doorEntities = mapData.doors.map(d => new DoorEntity(d.x, d.y));
    // Build door position lookup for group detection
    let doorByPos = {};
    for (let di = 0; di < this.doorEntities.length; di++) {
      let d = this.doorEntities[di];
      doorByPos[d.tileX + ',' + d.tileY] = d;
    }
    // Flood-fill to find orthogonally connected door groups
    let visited = {};
    for (let di2 = 0; di2 < this.doorEntities.length; di2++) {
      let startDoor = this.doorEntities[di2];
      let key = startDoor.tileX + ',' + startDoor.tileY;
      if (visited[key]) continue;
      let component = [];
      let queue = [startDoor];
      visited[key] = true;
      while (queue.length > 0) {
        let current = queue.shift();
        component.push(current);
        let dirs = [[0, -1], [0, 1], [-1, 0], [1, 0]];
        for (let di3 = 0; di3 < dirs.length; di3++) {
          let nx = current.tileX + dirs[di3][0];
          let ny = current.tileY + dirs[di3][1];
          let nkey = nx + ',' + ny;
          let neighbor = doorByPos[nkey];
          if (neighbor && !visited[nkey]) {
            visited[nkey] = true;
            queue.push(neighbor);
          }
        }
      }
      // Assign groupMembers if component has 2+ doors
      if (component.length >= 2) {
        for (let ci = 0; ci < component.length; ci++) {
          component[ci].groupMembers = component;
        }
      }
    }
    this.artifactEntities = [];
    this.waypoints = mapData.waypoints;
    this.altarPos = mapData.altarPos;
    this.playerSpawn = mapData.playerSpawn || { x: 1, y: 1 };
    this.monsterSpawn = mapData.monsterSpawn || mapData.altarPos || { x: 2, y: 2 };
    this.artifactEntities = [];
    this.randomMonsterSpawnPoints = mapData.randomSpawns || [];
    // Collect all F tile positions for dynamic spawn (artifacts spawn on monster summon)
    this.artifactFPoints = [];
    for (let y = 0; y < this.mapHeight; y++) {
      for (let x = 0; x < this.mapWidth; x++) {
        if (this.tiles[y][x] === 'F') this.artifactFPoints.push({ x, y });
      }
    }
    // Set artifact count from selected difficulty
    this.artifactCount = this.difficulty === 'easy' ? 3 : (this.difficulty === 'medium' ? 4 : 5);
    if (this.artifactFPoints.length < this.artifactCount) this.artifactCount = this.artifactFPoints.length;
    if (this.artifactCount === 0) this.artifactCount = 3;
    // Visibility
    this.visibilityGrid = [];
    this.visibilityAge = [];
    for (let y = 0; y < this.mapHeight; y++) {
      this.visibilityGrid.push(new Array(this.mapWidth).fill(0));
      this.visibilityAge.push(new Array(this.mapWidth).fill(0));
    }
    // Resize canvas
    this.canvas.width = this.mapWidth * TILE_SIZE;
    this.canvas.height = this.mapHeight * TILE_SIZE;
    // Create/reset players
    this.players = [];
    const count = this.playerCount;
    for (let i = 0; i < count; i++) {
      const ox = i > 0 ? ((i % 2 === 1 ? 1 : -1) * Math.ceil(i / 2)) : 0;
      const sx = Math.max(0, Math.min(this.mapWidth - 1, this.playerSpawn.x + ox));
      const sy = this.playerSpawn.y;
      const p = new Player(sx, sy, i, this.playerNames[i] || ('P' + (i + 1)), this.playerClasses[i] || 'none');
      if (i === 0) {
        p.px = this.playerSpawn.x * TILE_SIZE + TILE_SIZE / 2;
        p.py = this.playerSpawn.y * TILE_SIZE + TILE_SIZE / 2;
        p.tileX = this.playerSpawn.x;
        p.tileY = this.playerSpawn.y;
      }
      p.alive = true;
      p.hp = p.maxHp;
      p.interacting = false;
      p.interactTimer = 0;
      p.invulnTimer = 0;
      p.abilityCooldown = 0;
      p.abilityActive = false;
      p.abilityTimer = 0;
      if (p.playerClass === 'scout') p.speed = p.baseSpeed;
      p.inventory = [null, null, null, null, null];
      p.selectedSlot = 0;
      p.useTimer = 0;
      p.using = false;
      this.players.push(p);
    }
    // Clear monsters
    this.activeMonsters = [];
    this.monsterSpawnType = 'normal';
    this.summonPhase = '';
    this.summonPhaseTimer = 0;
    this.summonPhaseMessage = '';
    this.monsterDesc = '';
    // Reset state
    this.summonCount = 0;
    this.roundNumber = 0;
    this.collectedArtifacts = 0;
    this.isNight = false;
    this.nightTransition = 0;
    this.roofAlpha = 1;
    this.targetRoofAlpha = 1;
    this.summoning = false;
    this.summonTimer = 0;
    this.summonDelay = 0;
    this.particles = [];
    this.traps = [];
    // Resolve base tile type for special markers (F, U, m)
    this.baseTileResolved = new Array(this.mapHeight);
    for (let y = 0; y < this.mapHeight; y++) {
      this.baseTileResolved[y] = new Array(this.mapWidth);
      for (let x = 0; x < this.mapWidth; x++) {
        this.baseTileResolved[y][x] = this.resolveBaseTile(x, y);
      }
    }
    this.buildAutotileCache();
    this.initShops();
    this.setDifficulty();
    updateMapEditor(this);
    this.showMessage('');
  }

  selectArtifactsForRound() {
    // Select N unique random F points for this round
    const needed = this.artifactCount;
    const pool = [...this.artifactFPoints];
    const selected = [];
    while (selected.length < needed && pool.length > 0) {
      const idx = Math.floor(Math.random() * pool.length);
      selected.push(pool[idx]);
      pool.splice(idx, 1);
    }
    // Reset all existing entities
    for (const art of this.artifactEntities) art.collected = false;
    // Ensure we have exactly needed entities at the right positions
    this.artifactEntities = [];
    for (const sp of selected) {
      this.artifactEntities.push(new ArtifactEntity(sp.x, sp.y));
    }
  }

  getMonsterSpeed() {
    let bonus = 0;
    if (this.difficulty === 'easy') bonus = 5;
    else if (this.difficulty === 'medium') bonus = 10;
    else bonus = 20;
    return this.baseMonsterSpeed * (1 + bonus * this.roundNumber / 100);
  }

  getSpeedBonusPercent() {
    if (this.difficulty === 'easy') return this.roundNumber * 5;
    else if (this.difficulty === 'medium') return this.roundNumber * 10;
    else return this.roundNumber * 20;
  }

  setDifficulty() {
    if (this.artifactCount <= 3) this.difficulty = 'easy';
    else if (this.artifactCount <= 4) this.difficulty = 'medium';
    else this.difficulty = 'hard';
  }

  buildAutotileCache() {
    const h = this.mapHeight, w = this.mapWidth;
    this.tileVariation = new Array(h);
    this.tileEdgeMask = new Array(h);
    for (let y = 0; y < h; y++) {
      this.tileVariation[y] = new Float32Array(w);
      this.tileEdgeMask[y] = new Int8Array(w);
      for (let x = 0; x < w; x++) {
        this.tileVariation[y][x] = Game.tileHash(x, y);
        const resolved = this.baseTileResolved[y] ? this.baseTileResolved[y][x] : this.tiles[y][x];
        if (resolved !== 'C' && resolved !== 'G') {
          this.tileEdgeMask[y][x] = 0;
          continue;
        }
        let mask = 0;
        if (y > 0) {
          const up = this.baseTileResolved[y - 1] ? this.baseTileResolved[y - 1][x] : this.tiles[y - 1][x];
          if (up !== resolved) mask |= 1;
        }
        if (y < h - 1) {
          const down = this.baseTileResolved[y + 1] ? this.baseTileResolved[y + 1][x] : this.tiles[y + 1][x];
          if (down !== resolved) mask |= 2;
        }
        if (x > 0) {
          const left = this.baseTileResolved[y] ? this.baseTileResolved[y][x - 1] : this.tiles[y][x - 1];
          if (left !== resolved) mask |= 4;
        }
        if (x < w - 1) {
          const right = this.baseTileResolved[y] ? this.baseTileResolved[y][x + 1] : this.tiles[y][x + 1];
          if (right !== resolved) mask |= 8;
        }
        this.tileEdgeMask[y][x] = mask;
      }
    }
  }

  loop(now) {
    const dt = Math.min((now - this.lastTime) / 1000, 0.1);
    this.lastTime = now;
    this.update(dt);
    this.render();
    requestAnimationFrame(t => this.loop(t));
  }

  update(dt) {
    // Update doors
    for (const door of this.doorEntities) door.update(dt);
    // Push entities out of solid doors
    this.pushEntitiesFromDoors();
    // Update players
    for (const p of this.players) p.update(dt, this);
    // Update monsters
    for (const m of this.activeMonsters) m.update(dt);
    // Update traps
    this.updateTraps(dt);
    // Update artifacts
    for (const art of this.artifactEntities) art.update(dt);
    // Update particles
    for (let i = this.particles.length - 1; i >= 0; i--) {
      this.particles[i].update(dt);
      if (!this.particles[i].alive) this.particles.splice(i, 1);
    }
    // Update message
    if (this.messageTimer > 0) { this.messageTimer -= dt; if (this.messageTimer <= 0) this.showMessage(''); }
    // Summon phase sequencial
    if (this.summonPhase === 'name') {
      this.summonPhaseTimer -= dt;
      if (this.summonPhaseTimer <= 0) {
        this.summonPhase = 'desc';
        this.summonPhaseTimer = 3.0;
        this.showMessage('');
        this.showSummonDesc(this.monsterDesc);
      }
    } else if (this.summonPhase === 'desc') {
      this.summonPhaseTimer -= dt;
      if (this.summonPhaseTimer <= 0) {
        this.summonPhase = '';
        this.hideSummonDesc();
        this.monsterDesc = '';
        this.summonPhaseMessage = '';
      }
    }
    // Day/night transition
    const anyMonsterAlive = this.activeMonsters.some(m => m.alive);
    this.targetRoofAlpha = this.computeRoofAlpha();
    this.roofAlpha += (this.targetRoofAlpha - this.roofAlpha) * Math.min(dt * 4, 1);
    if (anyMonsterAlive && !this.isNight) {
      this.nightTransition = Math.min(1, this.nightTransition + dt * 0.8);
      if (this.nightTransition >= 1) this.isNight = true;
    } else if (!anyMonsterAlive && this.isNight) {
      this.nightTransition = Math.max(0, this.nightTransition - dt * 1.2);
      if (this.nightTransition <= 0) this.isNight = false;
    } else if (!anyMonsterAlive && !this.isNight && this.nightTransition > 0) {
      this.nightTransition = Math.max(0, this.nightTransition - dt * 1.2);
    }
    // Summon delay
    if (this.summonDelay > 0) {
      this.summonDelay -= dt;
      if (this.summonDelay <= 0) {
        this.spawnMonster();
      }
    }
    // Check game state
    this.checkGameState();
    // Update player vision ranges (indoor/outdoor)
    this.updateVisionRanges(dt);
    // Camera
    this.updateCamera();
    // Update HUD
    updateHUD(this);
    updateAbilityHUD(this);
    updateHotbarHUD(this);
    // Consume justPressed
    for (const k in keysJustPressed) keysJustPressed[k] = false;
  }

  computeRoofAlpha() {
    if (this.debugMode) return 0;
    // If any player is inside, no roof
    for (const pl of this.players) {
      if (!pl.alive) continue;
      const pt = pl.tileX;
      const pty = pl.tileY;
      const tile = (pty >= 0 && pty < this.mapHeight && pt >= 0 && pt < this.mapWidth)
        ? this.tiles[pty][pt] : 'G';
      const effectiveTile = (pty >= 0 && pty < this.mapHeight && pt >= 0 && pt < this.mapWidth)
        ? (this.baseTileResolved[pty] ? this.baseTileResolved[pty][pt] : tile) : 'G';
      if (effectiveTile !== 'G' && effectiveTile !== 'O') return 0;
    }
    // All alive players outside - check near open doors
    for (const pl of this.players) {
      if (!pl.alive) continue;
      const pt = pl.tileX;
      const pty = pl.tileY;
      for (let di = 0; di < this.doorEntities.length; di++) {
        const door = this.doorEntities[di];
        if (!door.open && !door.animating) continue;
        if (door.hasGroup()) {
          // For grouped doors, check distance to group bounding box
          const members = door.groupMembers;
          let gMinX = Infinity, gMaxX = -Infinity, gMinY = Infinity, gMaxY = -Infinity;
          for (let gi = 0; gi < members.length; gi++) {
            if (members[gi].tileX < gMinX) gMinX = members[gi].tileX;
            if (members[gi].tileX > gMaxX) gMaxX = members[gi].tileX;
            if (members[gi].tileY < gMinY) gMinY = members[gi].tileY;
            if (members[gi].tileY > gMaxY) gMaxY = members[gi].tileY;
          }
          // Manhattan distance to nearest edge of bounding box
          const ddx = pt < gMinX ? gMinX - pt : (pt > gMaxX ? pt - gMaxX : 0);
          const ddy = pty < gMinY ? gMinY - pty : (pty > gMaxY ? pty - gMaxY : 0);
          const dist = Math.sqrt(ddx * ddx + ddy * ddy);
          if (dist < 3) return Math.max(0, (dist - 1) / 2);
        } else {
          const dx = door.tileX - pt;
          const dy = door.tileY - pty;
          const dist = Math.sqrt(dx * dx + dy * dy);
          if (dist < 3) return Math.max(0, (dist - 1) / 2);
        }
      }
    }
    return 1;
  }

  pushEntitiesFromDoors() {
    for (const door of this.doorEntities) {
      if (!door.solid) continue;
      for (const p of this.players) {
        if (!p.alive) continue;
        this.pushOut(p, door.tileX, door.tileY);
      }
      for (const m of this.activeMonsters) {
        if (!m.alive) continue;
        this.pushOut(m, door.tileX, door.tileY);
      }
    }
  }

  pushOut(entity, doorTX, doorTY) {
    const et = Math.floor(entity.px / TILE_SIZE);
    const ety = Math.floor(entity.py / TILE_SIZE);
    if (et !== doorTX || ety !== doorTY) return;
    const dirs = [[0, -1], [0, 1], [-1, 0], [1, 0]];
    for (const [dx, dy] of dirs) {
      const nx = doorTX + dx, ny = doorTY + dy;
      if (nx < 0 || ny < 0 || nx >= this.mapWidth || ny >= this.mapHeight) continue;
      const tile = this.tiles[ny][nx];
      if (tile === 'O' || tile === 'W') continue;
      if (tile === 'd') {
        const d = this.doorEntities.find(dr => dr.tileX === nx && dr.tileY === ny);
        if (d && d.solid) continue;
      }
      entity.px = nx * TILE_SIZE + TILE_SIZE / 2;
      entity.py = ny * TILE_SIZE + TILE_SIZE / 2;
      return;
    }
  }

  updateCamera() {
    const alive = this.players.filter(p => p.alive);
    if (alive.length === 0) return;
    const mapW = this.mapWidth * TILE_SIZE;
    const mapH = this.mapHeight * TILE_SIZE;
    if (alive.length === 1) {
      // Single player: follow player, full map visible
      const p = alive[0];
      const targetCamX = p.px - mapW / 2;
      const targetCamY = p.py - mapH / 2;
      this.camX += (targetCamX - this.camX) * 0.15;
      this.camY += (targetCamY - this.camY) * 0.15;
      this.camScale += (1 - this.camScale) * 0.15;
    } else {
      // Multiplayer: show entire map
      this.camX += (0 - this.camX) * 0.1;
      this.camY += (0 - this.camY) * 0.1;
      this.camScale += (1 - this.camScale) * 0.1;
    }
  }

  updateVisibility() {
    const vRadius = this.isNight ? 5 : 12;
    const fogDecay = this.isNight ? 0.15 : 0.4;
    if (this.debugMode) {
      for (let y = 0; y < this.mapHeight; y++)
        for (let x = 0; x < this.mapWidth; x++)
          this.visibilityGrid[y][x] = 2;
      return;
    }
    // Age all tiles
    for (let y = 0; y < this.mapHeight; y++) {
      for (let x = 0; x < this.mapWidth; x++) {
        if (this.visibilityAge[y][x] > 0) {
          this.visibilityAge[y][x] -= fogDecay * 0.02;
          if (this.visibilityAge[y][x] <= 0) {
            this.visibilityAge[y][x] = 0;
            this.visibilityGrid[y][x] = 0;
          } else if (this.visibilityGrid[y][x] === 2) {
            this.visibilityGrid[y][x] = 1;
          }
        } else if (this.visibilityGrid[y][x] > 0) {
          this.visibilityGrid[y][x] = 0;
        }
      }
    }
    // Check visible tiles for each alive player (union)
    for (const pl of this.players) {
      if (!pl.alive) continue;
      const pt = pl.tileX;
      const pty = pl.tileY;
      for (let dy = -vRadius; dy <= vRadius; dy++) {
        for (let dx = -vRadius; dx <= vRadius; dx++) {
          const tx = pt + dx, ty = pty + dy;
          if (tx < 0 || ty < 0 || tx >= this.mapWidth || ty >= this.mapHeight) continue;
          if (dx * dx + dy * dy <= vRadius * vRadius) {
            if (this.hasLineOfSight(pt, pty, tx, ty)) {
              this.visibilityGrid[ty][tx] = 2;
              this.visibilityAge[ty][tx] = 1;
            }
          }
        }
      }
    }
  }

  hasLineOfSight(x1, y1, x2, y2) {
    const dx = Math.abs(x2 - x1), dy = Math.abs(y2 - y1);
    const sx = x1 < x2 ? 1 : -1, sy = y1 < y2 ? 1 : -1;
    let err = dx - dy, cx = x1, cy = y1;
    const steps = dx + dy;
    for (let i = 0; i < steps; i++) {
      if (cx === x2 && cy === y2) break;
      const e2 = err * 2;
      if (e2 > -dy) { err -= dy; cx += sx; }
      if (e2 < dx) { err += dx; cy += sy; }
      if (cx >= 0 && cx < this.mapWidth && cy >= 0 && cy < this.mapHeight) {
        if (this.tiles[cy][cx] === 'O') return false;
        if (this.tiles[cy][cx] === 'd') {
          const door = this.doorEntities.find(d => d.tileX === cx && d.tileY === cy);
          if (!door || door.solid) return false;
        }
      }
    }
    return true;
  }

  findPath(sx, sy, gx, gy) {
    if (sx === gx && sy === gy) return [{ x: gx, y: gy }];
    const w = this.mapWidth, h = this.mapHeight;
    if (sx < 0 || sy < 0 || sx >= w || sy >= h ||
        gx < 0 || gy < 0 || gx >= w || gy >= h) return null;
    const visited = new Array(h);
    const parent = new Array(h);
    for (let i = 0; i < h; i++) {
      visited[i] = new Array(w).fill(false);
      parent[i] = new Array(w).fill(null);
    }
    const queue = [{ x: sx, y: sy }];
    visited[sy][sx] = true;
    let head = 0;
    const dirs = [[0, -1], [0, 1], [-1, 0], [1, 0]];
    while (head < queue.length) {
      const { x: cx, y: cy } = queue[head++];
      if (cx === gx && cy === gy) {
        const path = [];
        let bx = cx, by = cy;
        while (bx !== sx || by !== sy) {
          path.unshift({ x: bx, y: by });
          const p = parent[by][bx];
          if (!p) break;
          bx = p.x; by = p.y;
        }
        return path;
      }
      for (const [dx, dy] of dirs) {
        const nx = cx + dx, ny = cy + dy;
        if (nx < 0 || ny < 0 || nx >= w || ny >= h || visited[ny][nx]) continue;
        const tile = this.tiles[ny][nx];
        if (tile === 'O' || tile === 'W') continue;
        visited[ny][nx] = true;
        parent[ny][nx] = { x: cx, y: cy };
        queue.push({ x: nx, y: ny });
      }
    }
    return null;
  }

  // ============================================================
  // RAYCAST LIGHTING SYSTEM
  // ============================================================
  castRay(px, py, dx, dy, maxDist) {
    const step = 3;
    let x = px, y = py;
    let dist = 0;
    while (dist < maxDist) {
      dist += step;
      x += dx * step;
      y += dy * step;
      const tx = Math.floor(x / TILE_SIZE);
      const ty = Math.floor(y / TILE_SIZE);
      if (tx < 0 || ty < 0 || tx >= this.mapWidth || ty >= this.mapHeight) break;
      const tile = this.tiles[ty][tx];
      if (tile === 'O') {
        x -= dx * step;
        y -= dy * step;
        break;
      }
      if (tile === 'd') {
        const door = this.doorEntities.find(d => d.tileX === tx && d.tileY === ty);
        if (!door || door.solid) {
          x -= dx * step;
          y -= dy * step;
          break;
        }
      }
    }
    return { x, y };
  }

  updateVisionRanges(dt) {
    const indoorRange = Math.sqrt(this.mapWidth * this.mapWidth + this.mapHeight * this.mapHeight) * TILE_SIZE;
    const outdoorRange = 160; // 5 tiles
    const lerpSpeed = 3;
    for (let i = 0; i < this.players.length; i++) {
      const p = this.players[i];
      if (!p.alive) { this.playerVisionRange[i] = outdoorRange; continue; }
      const tx = p.tileX, ty = p.tileY;
      let target;
      if (tx < 0 || ty < 0 || tx >= this.mapWidth || ty >= this.mapHeight) {
        target = outdoorRange;
      } else {
        const rawTile = this.tiles[ty][tx];
        const effectiveTile = this.baseTileResolved[ty] ? this.baseTileResolved[ty][tx] : rawTile;
        target = (effectiveTile === 'G' || effectiveTile === 'O') ? outdoorRange : indoorRange;
      }
      if (this.playerVisionRange[i] === undefined) this.playerVisionRange[i] = target;
      this.playerVisionRange[i] += (target - this.playerVisionRange[i]) * Math.min(dt * lerpSpeed, 1);
    }
  }

  buildVisibilityPolygon(player) {
    const baseRange = this.isNight ? 160 : 384;
    const playerRange = this.playerVisionRange[player.playerIndex] || baseRange;
    const maxDist = playerRange;
    const numRays = this.isNight ? 360 : 220;
    const points = [];
    for (let i = 0; i < numRays; i++) {
      const angle = (i / numRays) * Math.PI * 2;
      const dx = Math.cos(angle);
      const dy = Math.sin(angle);
      const hit = this.castRay(player.px, player.py, dx, dy, maxDist);
      points.push(hit);
    }
    for (let i = 0; i < points.length; i++) {
      const a = (i / numRays) * Math.PI * 2;
      const jitter = 2 + Math.sin(i * 7.3) * 1.5;
      points[i] = {
        x: points[i].x + Math.cos(a) * jitter,
        y: points[i].y + Math.sin(a) * jitter
      };
    }
    return points;
  }

  buildLightmap() {
    const W = this.canvas.width;
    const H = this.canvas.height;
    if (!this.lightmapCanvas || this.lightmapCanvas.width !== W || this.lightmapCanvas.height !== H) {
      this.lightmapCanvas = document.createElement('canvas');
      this.lightmapCanvas.width = W;
      this.lightmapCanvas.height = H;
    }
    const lctx = this.lightmapCanvas.getContext('2d');
    const nightAlpha = 0.55 + this.nightTransition * 0.15;
    lctx.clearRect(0, 0, W, H);
    // Fill entire map with darkness
    lctx.fillStyle = 'rgba(0,0,0,' + nightAlpha + ')';
    lctx.fillRect(0, 0, W, H);

    this.visibilityPolygons = [];
    for (const pl of this.players) {
      if (!pl.alive) continue;
      const points = this.buildVisibilityPolygon(pl);
      if (points.length < 3) continue;
      this.visibilityPolygons.push({ player: pl, points: points });
      const sx = pl.px - this.camX;
      const sy = pl.py - this.camY;
      lctx.save();
      lctx.beginPath();
      lctx.moveTo(points[0].x - this.camX, points[0].y - this.camY);
      for (let i = 1; i < points.length; i++) {
        lctx.lineTo(points[i].x - this.camX, points[i].y - this.camY);
      }
      lctx.closePath();
      lctx.clip();
      // Erase darkness inside polygon using destination-out
      lctx.globalCompositeOperation = 'destination-out';
      const maxR = this.playerVisionRange[pl.playerIndex] || (this.isNight ? 160 : 384);
      const grad = lctx.createRadialGradient(sx, sy, 0, sx, sy, maxR);
      grad.addColorStop(0, 'rgba(0,0,0,1)');
      grad.addColorStop(0.25, 'rgba(0,0,0,0.85)');
      grad.addColorStop(0.5, 'rgba(0,0,0,0.55)');
      grad.addColorStop(0.75, 'rgba(0,0,0,0.2)');
      grad.addColorStop(1, 'rgba(0,0,0,0.0)');
      lctx.fillStyle = grad;
      lctx.fillRect(0, 0, W, H);
      lctx.restore();
    }
    return this.lightmapCanvas;
  }

  isPointVisible(px, py) {
    if (this.debugMode) return true;
    for (const vp of this.visibilityPolygons) {
      const pts = vp.points;
      let inside = false;
      for (let i = 0, j = pts.length - 1; i < pts.length; j = i++) {
        const xi = pts[i].x, yi = pts[i].y;
        const xj = pts[j].x, yj = pts[j].y;
        if (((yi > py) !== (yj > py)) && (px < (xj - xi) * (py - yi) / (yj - yi) + xi)) {
          inside = !inside;
        }
      }
      if (inside) return true;
    }
    return false;
  }

  checkGameState() {
    if (this.activeMonsters.length === 0 || this.activeMonsters.every(m => !m.alive)) return;
    let allCollected = true;
    for (const art of this.artifactEntities) {
      if (!art.collected) { allCollected = false; break; }
    }
    if (allCollected) {
      for (const m of this.activeMonsters) m.die();
      this.activeMonsters = [];
      this.summoning = false;
      this.collectedArtifacts = 0;
      this.isNight = false;
      for (const p of this.players) {
        if (!p.alive) {
          p.alive = true;
          p.hp = p.maxHp;
          p.px = this.playerSpawn.x * TILE_SIZE + TILE_SIZE / 2;
          p.py = this.playerSpawn.y * TILE_SIZE + TILE_SIZE / 2;
          p.tileX = this.playerSpawn.x;
          p.tileY = this.playerSpawn.y;
          p.invulnTimer = 1.5;
        }
      }
      this.showMessage('Round ' + this.roundNumber + ' complete! Monster defeated!');
      // Round reward: random item to all players
      const keys = Object.keys(Items);
      for (const p of this.players) {
        if (!p.alive) continue;
        if (p.hasSpace()) {
          const key = keys[Math.floor(Math.random() * keys.length)];
          p.addToInventory(key);
          this.showMessage(p.name + ' received ' + Items[key].name + '!');
        }
      }
      this.restockShops();
    }
  }

  respawnArtifacts() {
    // Find all artifact spawn points
    const spawnPoints = [];
    for (let y = 0; y < this.mapHeight; y++) {
      for (let x = 0; x < this.mapWidth; x++) {
        if (this.tiles[y][x] === 'F') spawnPoints.push({ x, y });
      }
    }
    // Reset existing artifacts
    for (const art of this.artifactEntities) art.collected = false;
    // If not enough entities, create more
    while (this.artifactEntities.length < this.artifactCount && spawnPoints.length > 0) {
      const sp = spawnPoints[Math.floor(Math.random() * spawnPoints.length)];
      this.artifactEntities.push(new ArtifactEntity(sp.x, sp.y));
    }
  }

  spawnMonster() {
    this.summonCount++;
    this.roundNumber++;
    const typeKey = weightedRandom(MonsterTypes);
    const mt = MonsterTypes[typeKey];
    this.monsterSpawnType = typeKey;
    this.summonPhase = 'name';
    this.summonPhaseTimer = 2.0;
    this.summonPhaseMessage = mt.name + ' ouviu voce. Corra.';
    this.monsterDesc = mt.desc;
    this.showMessage(this.summonPhaseMessage);
    const spd = this.getMonsterSpeed() * (mt.speedMod || 1);
    this.selectArtifactsForRound();
    // Determine spawn position
    let spawnX = this.monsterSpawn.x;
    let spawnY = this.monsterSpawn.y;
    if (mt.randomSpawn && this.randomMonsterSpawnPoints.length > 0) {
      const rp = this.randomMonsterSpawnPoints[Math.floor(Math.random() * this.randomMonsterSpawnPoints.length)];
      spawnX = rp.x;
      spawnY = rp.y;
    }
    for (let i = 0; i < mt.spawnCount; i++) {
      const ox = i > 0 ? (Math.random() - 0.5) * 2 : 0;
      const oy = i > 0 ? (Math.random() - 0.5) * 2 : 0;
      const stx = Math.round(spawnX + ox);
      const sty = Math.round(spawnY + oy);
      const m = new MonsterEntity(
        Math.max(0, Math.min(this.mapWidth - 1, stx)),
        Math.max(0, Math.min(this.mapHeight - 1, sty)),
        this, typeKey
      );
      m.speed = spd;
      m.spawn();
      this.activeMonsters.push(m);
    }
  }

  collectArtifact(art) {
    if (art.collected) return;
    art.collected = true;
    this.collectedArtifacts++;
    this.showMessage('Artifact collected! (' + this.collectedArtifacts + '/' + this.artifactCount + ')');
    for (let i = 0; i < 6; i++) {
      this.particles.push(new Particle(
        art.tileX * TILE_SIZE + TILE_SIZE / 2,
        art.tileY * TILE_SIZE + TILE_SIZE / 2,
        (Math.random() - 0.5) * 100, (Math.random() - 0.5) * 100,
        0.5, '#00cccc'
      ));
    }
  }

  updateInteraction(player) {
    const target = player.interactTarget;
    const type = player.interactType;
    if (!target || !type) return;
    let duration = 0;
    if (type === 'door') duration = 0.3;
    else if (type === 'altar') duration = 4;
    else if (type === 'artifact') duration = 1.5;
    this.interactDisplay = { type, duration, target };
    this.interactDuration = duration;
    if (player.interactTimer >= duration) {
      player.interacting = false;
      player.interactTimer = 0;
      if (type === 'door') target.toggle();
      else if (type === 'altar' && !this.summoning) {
        this.summoning = true;
        this.summonDelay = 3;
        this.showMessage('Summoning ritual begins...');
      }
      else if (type === 'artifact') this.collectArtifact(target);
      else if (type === 'shop') {
        this.buyFromShop(player, target);
      }
      else if (type === 'trash') {
        this.sellItem(player);
      }
    }
  }

  useItem(player) {
    const item = player.getEquippedItem();
    if (!item) return;
    const it = Items[item];
    if (!it || !it.stun) return;
    this.traps.push({ x: player.px, y: player.py, item: item, active: true, stun: it.stun, trapColor: it.trapColor, trapRadius: it.trapRadius });
    player.removeEquipped();
    this.showMessage(player.name + ' placed ' + it.name);
  }

  buyFromShop(player, shopData) {
    if (!shopData || shopData.sold) return;
    if (!player.hasSpace()) { this.showMessage(player.name + "'s inventory is full."); return; }
    if (this.currency < shopData.price) { this.showMessage('Not enough money (' + this.currency + ' coins).'); return; }
    this.currency -= shopData.price;
    player.addToInventory(shopData.itemKey);
    shopData.sold = true;
    this.showMessage(player.name + ' bought ' + Items[shopData.itemKey].name);
  }

  sellItem(player) {
    const item = player.getEquippedItem();
    if (!item) { this.showMessage(player.name + ' has no item to sell.'); return; }
    const price = Items[item].price;
    this.currency += price;
    player.removeEquipped();
    this.showMessage(player.name + ' sold ' + Items[item].name + ' for ' + price + ' coins.');
  }

  initShops() {
    this.shopItems = [];
    for (let y = 0; y < this.mapHeight; y++) {
      for (let x = 0; x < this.mapWidth; x++) {
        if (this.tiles[y][x] !== 'S') continue;
        const keys = Object.keys(Items);
        const key = keys[Math.floor(Math.random() * keys.length)];
        this.shopItems.push({
          x, y, itemKey: key, price: Items[key].price, sold: false,
          worldX: x * TILE_SIZE + TILE_SIZE / 2,
          worldY: y * TILE_SIZE + TILE_SIZE / 2
        });
      }
    }
  }

  restockShops() {
    for (const shop of this.shopItems) {
      shop.sold = false;
      const keys = Object.keys(Items);
      const key = keys[Math.floor(Math.random() * keys.length)];
      shop.itemKey = key;
      shop.price = Items[key].price;
    }
  }

  updateTraps(dt) {
    for (const m of this.activeMonsters) {
      if (!m.alive || m.stunTimer > 0) continue;
      for (let i = this.traps.length - 1; i >= 0; i--) {
        const t = this.traps[i];
        if (!t.active) continue;
        const dx = m.px - t.x;
        const dy = m.py - t.y;
        if (Math.sqrt(dx * dx + dy * dy) < TILE_SIZE * 0.5) {
          m.stunTimer = t.stun;
          t.active = false;
          this.traps.splice(i, 1);
          this.showMessage(Items[t.item].name + ' triggered! Monster stunned ' + t.stun + 's');
          break;
        }
      }
    }
  }

  renderTraps(ctx) {
    for (const t of this.traps) {
      if (!t.active) continue;
      if (!this.isPointVisible(t.x, t.y)) continue;
      const sx = t.x - this.camX;
      const sy = t.y - this.camY;
      if (sx < -50 || sx > this.canvas.width + 50 || sy < -50 || sy > this.canvas.height + 50) continue;
      ctx.fillStyle = t.trapColor;
      ctx.beginPath();
      ctx.arc(sx, sy, t.trapRadius, 0, Math.PI * 2);
      ctx.fill();
      ctx.strokeStyle = '#fff';
      ctx.lineWidth = 0.5;
      ctx.stroke();
      // Cross mark
      ctx.strokeStyle = 'rgba(255,255,255,0.5)';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(sx - 3, sy);
      ctx.lineTo(sx + 3, sy);
      ctx.moveTo(sx, sy - 3);
      ctx.lineTo(sx, sy + 3);
      ctx.stroke();
    }
  }

  renderShops(ctx) {
    if (this.activeMonsters.some(m => m.alive)) return;
    for (const shop of this.shopItems) {
      if (shop.sold) continue;
      const sx = shop.worldX - this.camX;
      const sy = shop.worldY - this.camY;
      if (sx < -50 || sx > this.canvas.width + 50 || sy < -50 || sy > this.canvas.height + 50) continue;
      const it = Items[shop.itemKey];
      // Pedestal
      ctx.fillStyle = '#3a3030';
      ctx.fillRect(sx - 10, sy - 10, 20, 20);
      ctx.strokeStyle = '#888';
      ctx.lineWidth = 1;
      ctx.strokeRect(sx - 10, sy - 10, 20, 20);
      // Item icon
      ctx.fillStyle = it.color;
      ctx.font = 'bold 14px "Courier New", monospace';
      ctx.textAlign = 'center';
      ctx.fillText(it.icon, sx, sy + 5);
      ctx.textAlign = 'start';
      // Price
      ctx.fillStyle = '#ff0';
      ctx.font = '9px "Courier New", monospace';
      ctx.fillText(it.price + 'g', sx - 8, sy - 14);
    }
  }

  renderTrashBins(ctx) {
    if (this.activeMonsters.some(m => m.alive)) return;
    for (let y = 0; y < this.mapHeight; y++) {
      for (let x = 0; x < this.mapWidth; x++) {
        if (this.tiles[y][x] !== 'T') continue;
        const sx = x * TILE_SIZE + TILE_SIZE / 2 - this.camX;
        const sy = y * TILE_SIZE + TILE_SIZE / 2 - this.camY;
        if (sx < -50 || sx > this.canvas.width + 50 || sy < -50 || sy > this.canvas.height + 50) continue;
        ctx.fillStyle = '#555';
        ctx.fillRect(sx - 8, sy - 10, 16, 20);
        ctx.strokeStyle = '#888';
        ctx.lineWidth = 1;
        ctx.strokeRect(sx - 8, sy - 10, 16, 20);
        ctx.fillStyle = '#aaa';
        ctx.font = '9px "Courier New", monospace';
        ctx.textAlign = 'center';
        ctx.fillText('$', sx, sy + 3);
        ctx.textAlign = 'start';
      }
    }
  }

  // ============================================================
  // RENDERING
  // ============================================================
  render() {
    const ctx = this.ctx;
    const W = this.canvas.width, H = this.canvas.height;
    ctx.clearRect(0, 0, W, H);
    // Apply camera scale
    ctx.save();
    ctx.translate(W / 2, H / 2);
    ctx.scale(this.camScale, this.camScale);
    ctx.translate(-W / 2, -H / 2);
    // Background
    ctx.fillStyle = '#111';
    ctx.fillRect(0, 0, W, H);
    // Draw tiles
    this.renderTiles(ctx);
    // Draw artifacts on floor
    this.renderArtifacts(ctx);
    // Draw roof overlay
    this.renderRoof(ctx);
    // Raycast lightmap (replaces old fog)
    const lightmap = this.buildLightmap();
    ctx.drawImage(lightmap, 0, 0);
    // Doors on top of lightmap (always lit when near player)
    this.renderDoors(ctx);
    // Traps on top of doors (with visibility check)
    this.renderTraps(ctx);
    // Shops and trash bins on top of lightmap
    this.renderShops(ctx);
    this.renderTrashBins(ctx);
    // Draw entities on top
    for (const p of this.players) p.render(ctx, this);
    for (const m of this.activeMonsters) m.render(ctx, this);
    this.renderParticles(ctx);
    // Interaction progress circle (canvas)
    this.renderInteractionCircle(ctx);
    // Night overlay
    this.renderNight(ctx);
    // Debug
    if (this.debugMode) this.renderDebug(ctx);
    ctx.restore();
    // Message (DOM)
    this.renderMessage(ctx);
    // Interaction bar (DOM)
    this.renderInteractionBar();
  }

  renderTiles(ctx) {
    for (let y = 0; y < this.mapHeight; y++) {
      for (let x = 0; x < this.mapWidth; x++) {
        const tile = this.tiles[y][x];
        const sx = x * TILE_SIZE - this.camX;
        const sy = y * TILE_SIZE - this.camY;
        if (sx + TILE_SIZE < -50 || sx > this.canvas.width + 50 || sy + TILE_SIZE < -50 || sy > this.canvas.height + 50) continue;
        let color = '#111';
        switch (tile) {
          case 'O': color = WALL_COLOR; break;
          case 'C': color = INTERIOR_COLOR; break;
          case 'G': color = GRASS_COLOR; break;
          case 'd': {
            const door = this.doorEntities.find(d => d.tileX === x && d.tileY === y);
            if (door && door.hasGroup()) {
              color = DOOR_CLOSED_COLOR;
            } else {
              color = (door && door.open) ? DOOR_OPEN_COLOR : DOOR_CLOSED_COLOR;
            }
            break;
          }
          case 'A': color = ALTAR_COLOR; break;
          case 'P': case 'M': color = INTERIOR_COLOR; break;
          case 'F': case 'U': case 'm': {
            const resolved = this.baseTileResolved[y] ? this.baseTileResolved[y][x] : 'C';
            color = (resolved === 'G') ? GRASS_COLOR : INTERIOR_COLOR;
            break;
          }
          case 'W': color = FURNITURE_COLOR; break;
          case 'S': case 'T': color = INTERIOR_COLOR; break;
        }
        ctx.fillStyle = color;
        ctx.fillRect(sx, sy, TILE_SIZE, TILE_SIZE);

        const effTile = (tile === 'F' || tile === 'U' || tile === 'm')
          ? (this.baseTileResolved[y] ? this.baseTileResolved[y][x] : 'C')
          : tile;

        // Autotiling: floor variations (C, P, M, S, T + F/U/m resolved as C)
        if (effTile === 'C' || tile === 'P' || tile === 'M' || tile === 'S' || tile === 'T') {
          const v = (this.tileVariation[y] && this.tileVariation[y][x]) || 0.5;
          const mask = (this.tileEdgeMask[y] && this.tileEdgeMask[y][x]) || 0;
          // Subtle plank lines
          ctx.fillStyle = 'rgba(0,0,0,0.04)';
          const plankOff = (v * 6) | 0;
          for (let py = plankOff; py < TILE_SIZE; py += 6) {
            ctx.fillRect(sx, sy + py, TILE_SIZE, 1);
          }
          // Edge darkening near different tile types
          if (mask) {
            ctx.fillStyle = 'rgba(0,0,0,0.06)';
            if (mask & 1) ctx.fillRect(sx, sy, TILE_SIZE, 2);
            if (mask & 2) ctx.fillRect(sx, sy + TILE_SIZE - 2, TILE_SIZE, 2);
            if (mask & 4) ctx.fillRect(sx, sy, 2, TILE_SIZE);
            if (mask & 8) ctx.fillRect(sx + TILE_SIZE - 2, sy, 2, TILE_SIZE);
          }
        }

        // Grass autotiling (G tiles + F/U/m resolved as G)
        if (effTile === 'G') {
          const v = (this.tileVariation[y] && this.tileVariation[y][x]) || 0.5;
          const mask = (this.tileEdgeMask[y] && this.tileEdgeMask[y][x]) || 0;
          // Color variation: shift hue slightly
          const r = 58 + (v - 0.5) * 14;
          const g = 122 + (v - 0.5) * 20;
          const b = 42 + (v - 0.5) * 10;
          ctx.fillStyle = 'rgb(' + (r | 0) + ',' + (g | 0) + ',' + (b | 0) + ')';
          ctx.fillRect(sx, sy, TILE_SIZE, TILE_SIZE);
          // Subtle grass blade details
          ctx.fillStyle = 'rgba(255,255,255,0.02)';
          const detailCount = 3 + ((v * 10) | 0) % 4;
          for (let i = 0; i < detailCount; i++) {
            const dx = sx + 4 + ((Game.tileHash(x * 3 + i, y * 3) * (TILE_SIZE - 8)) | 0);
            const dy = sy + 4 + ((Game.tileHash(x * 5 + i, y * 5 + 1) * (TILE_SIZE - 8)) | 0);
            ctx.fillRect(dx, dy, 2, 2);
          }
          // Edge transition near different tile types
          if (mask) {
            ctx.fillStyle = 'rgba(0,0,0,0.04)';
            if (mask & 1) ctx.fillRect(sx, sy, TILE_SIZE, 2);
            if (mask & 2) ctx.fillRect(sx, sy + TILE_SIZE - 2, TILE_SIZE, 2);
            if (mask & 4) ctx.fillRect(sx, sy, 2, TILE_SIZE);
            if (mask & 8) ctx.fillRect(sx + TILE_SIZE - 2, sy, 2, TILE_SIZE);
          }
        }

        // Waypoint/artifact overlay marks (drawn on top of grass/floor)
        if (tile === 'U') {
          ctx.fillStyle = 'rgba(0,0,0,0.08)';
          ctx.fillRect(sx + 4, sy + 4, TILE_SIZE - 8, TILE_SIZE - 8);
          ctx.fillStyle = 'rgba(0,0,0,0.04)';
          ctx.beginPath();
          ctx.arc(sx + TILE_SIZE / 2, sy + TILE_SIZE / 2, TILE_SIZE * 0.22, 0, Math.PI * 2);
          ctx.fill();
        }
        if (tile === 'F') {
          ctx.fillStyle = 'rgba(0,255,255,0.03)';
          ctx.fillRect(sx + 6, sy + 6, TILE_SIZE - 12, TILE_SIZE - 12);
        }

        // Grid lines in debug
        if (this.debugMode) {
          ctx.strokeStyle = 'rgba(255,255,255,0.1)';
          ctx.lineWidth = 0.5;
          ctx.strokeRect(sx, sy, TILE_SIZE, TILE_SIZE);
          if (tile === 'G' || tile === 'C') {
            ctx.fillStyle = '#0f0';
            ctx.font = '7px monospace';
            const m = (this.tileEdgeMask[y] && this.tileEdgeMask[y][x]) || 0;
            ctx.fillText(m.toString(16), sx + 2, sy + 10);
          }
        }
        // Draw door indicator
        if (tile === 'd') {
          const door = this.doorEntities.find(d => d.tileX === x && d.tileY === y);
          if (door && door.hasGroup()) {
            // Grouped doors are rendered by renderDoors, skip individual indicator
          } else if (door && !door.open) {
            ctx.fillStyle = '#8b6914';
            ctx.fillRect(sx + TILE_SIZE * 0.1, sy + TILE_SIZE * 0.1, TILE_SIZE * 0.8, TILE_SIZE * 0.8);
          } else if (door && door.animating) {
            const progress = door.animProgress;
            ctx.fillStyle = '#8b6914';
            const barW = TILE_SIZE * 0.8 * progress;
            ctx.fillRect(sx + TILE_SIZE * 0.1, sy + TILE_SIZE * 0.1, barW, TILE_SIZE * 0.8);
          }
        }
        // Draw altar marker
        if (tile === 'A') {
          ctx.fillStyle = '#ff4444';
          ctx.fillRect(sx + TILE_SIZE * 0.15, sy + TILE_SIZE * 0.15, TILE_SIZE * 0.7, TILE_SIZE * 0.7);
          ctx.strokeStyle = '#aa2222';
          ctx.lineWidth = 2;
          ctx.strokeRect(sx + TILE_SIZE * 0.15, sy + TILE_SIZE * 0.15, TILE_SIZE * 0.7, TILE_SIZE * 0.7);
        }
        // Draw furniture
        if (tile === 'W') {
          ctx.fillStyle = '#666';
          ctx.fillRect(sx + TILE_SIZE * 0.15, sy + TILE_SIZE * 0.15, TILE_SIZE * 0.7, TILE_SIZE * 0.7);
        }
      }
    }
  }

  renderArtifacts(ctx) {
    for (const art of this.artifactEntities) {
      if (art.collected) continue;
      art.render(ctx, art.tileX, art.tileY, TILE_SIZE, this.camX, this.camY);
    }
  }

  renderRoof(ctx) {
    if (this.roofAlpha <= 0.01) return;
    const alpha = this.roofAlpha;
    ctx.fillStyle = `rgba(30,20,15,${alpha * 0.85})`;
    for (let y = 0; y < this.mapHeight; y++) {
      for (let x = 0; x < this.mapWidth; x++) {
        const rawTile = this.tiles[y][x];
        const effectiveTile = this.baseTileResolved[y] ? this.baseTileResolved[y][x] : rawTile;
        if (effectiveTile === 'G' || effectiveTile === 'O') continue; // Exterior
        const sx = x * TILE_SIZE - this.camX;
        const sy = y * TILE_SIZE - this.camY;
        // Check if near open door
        let nearOpenDoor = false;
        for (let di = 0; di < this.doorEntities.length; di++) {
          const door = this.doorEntities[di];
          if (!door.open && !door.animating) continue;
          if (door.hasGroup()) {
            // For grouped doors, check if tile is near the group bounding box
            const members = door.groupMembers;
            let gMinX = Infinity, gMaxX = -Infinity, gMinY = Infinity, gMaxY = -Infinity;
            for (let gi = 0; gi < members.length; gi++) {
              if (members[gi].tileX < gMinX) gMinX = members[gi].tileX;
              if (members[gi].tileX > gMaxX) gMaxX = members[gi].tileX;
              if (members[gi].tileY < gMinY) gMinY = members[gi].tileY;
              if (members[gi].tileY > gMaxY) gMaxY = members[gi].tileY;
            }
            if (x >= gMinX - 2 && x <= gMaxX + 2 && y >= gMinY - 2 && y <= gMaxY + 2 && door.open) nearOpenDoor = true;
          } else {
            const dx = door.tileX - x;
            const dy = door.tileY - y;
            if (Math.abs(dx) <= 2 && Math.abs(dy) <= 2 && door.open) nearOpenDoor = true;
          }
          if (nearOpenDoor) break;
        }
        if (nearOpenDoor) {
          ctx.fillStyle = `rgba(30,20,15,${alpha * 0.3})`;
        } else {
          ctx.fillStyle = `rgba(30,20,15,${alpha * 0.85})`;
        }
        ctx.fillRect(sx, sy, TILE_SIZE, TILE_SIZE);
      }
    }
  }

  renderParticles(ctx) {
    for (const p of this.particles) {
      p.render(ctx, this.camX, this.camY);
    }
  }

  renderFog(ctx) {
    for (let y = 0; y < this.mapHeight; y++) {
      for (let x = 0; x < this.mapWidth; x++) {
        const vis = this.visibilityGrid[y][x];
        if (vis >= 2) continue;
        const sx = x * TILE_SIZE - this.camX;
        const sy = y * TILE_SIZE - this.camY;
        if (vis === 1) {
          ctx.fillStyle = 'rgba(0,0,0,0.6)';
        } else {
          ctx.fillStyle = 'rgba(0,0,0,0.92)';
        }
        ctx.fillRect(sx, sy, TILE_SIZE, TILE_SIZE);
      }
    }
  }

  renderDoors(ctx) {
    const ts = TILE_SIZE;
    const doorRange = 160;
    const renderedGroups = new Set();
    for (let di = 0; di < this.doorEntities.length; di++) {
      const door = this.doorEntities[di];
      // Grouped doors: render entire group once using the first door in the group
      if (door.groupMembers) {
        if (renderedGroups.has(door)) continue;
        for (let gi = 0; gi < door.groupMembers.length; gi++) {
          renderedGroups.add(door.groupMembers[gi]);
        }
        // Compute group bounding box
        let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
        for (let gi = 0; gi < door.groupMembers.length; gi++) {
          const gm = door.groupMembers[gi];
          if (gm.tileX < minX) minX = gm.tileX;
          if (gm.tileY < minY) minY = gm.tileY;
          if (gm.tileX > maxX) maxX = gm.tileX;
          if (gm.tileY > maxY) maxY = gm.tileY;
        }
        const gw = maxX - minX + 1;
        const gh = maxY - minY + 1;
        const gsx = minX * ts - this.camX;
        const gsy = minY * ts - this.camY;
        const gsw = gw * ts;
        const gsh = gh * ts;
        if (gsx + gsw < -50 || gsx > this.canvas.width + 50 || gsy + gsh < -50 || gsy > this.canvas.height + 50) continue;
        // Proximity check using group center
        const gcx = minX * ts + gsw / 2;
        const gcy = minY * ts + gsh / 2;
        let nearGroup = false;
        for (let pi = 0; pi < this.players.length; pi++) {
          const pl = this.players[pi];
          if (!pl.alive) continue;
          const pdx = pl.px - gcx;
          const pdy = pl.py - gcy;
          if (Math.sqrt(pdx * pdx + pdy * pdy) < doorRange + Math.max(gsw, gsh) * 0.5) { nearGroup = true; break; }
        }
        if (!nearGroup) continue;
        // Fill background for all tiles in the group
        ctx.fillStyle = INTERIOR_COLOR;
        ctx.fillRect(gsx, gsy, gsw, gsh);
        // Determine primary door state (use first door's state - all sync via toggle)
        const primary = door.groupMembers[0];
        const anyClosed = primary.open === false && primary.animating === false;
        const anyAnimating = primary.animating;
        const progress = primary.animProgress;
        const isHorizontal = gw > gh;
        if (anyAnimating) {
          ctx.fillStyle = DOOR_CLOSED_COLOR;
          if (isHorizontal) {
            // Horizontal group: door slides up (height shrinks)
            const doorH = gsh * 0.7 * progress;
            const doorY = gsy + gsh * 0.15;
            ctx.fillRect(gsx + gsw * 0.075, doorY, gsw * 0.85, doorH);
          } else {
            // Vertical group: door slides to the side (width shrinks)
            const doorW = gsw * 0.7 * progress;
            const doorX = gsx + gsw * 0.15;
            ctx.fillRect(doorX, gsy + gsh * 0.075, doorW, gsh * 0.85);
          }
        } else if (anyClosed) {
          ctx.fillStyle = DOOR_CLOSED_COLOR;
          if (isHorizontal) {
            const doorH = gsh * 0.7;
            const doorY = gsy + gsh * 0.15;
            ctx.fillRect(gsx + gsw * 0.075, doorY, gsw * 0.85, doorH);
          } else {
            const doorW = gsw * 0.7;
            const doorX = gsx + gsw * 0.15;
            ctx.fillRect(doorX, gsy + gsh * 0.075, doorW, gsh * 0.85);
          }
        }
        // Door handle (only when closed)
        if (anyClosed) {
          ctx.fillStyle = '#ffd700';
          ctx.beginPath();
          ctx.arc(gcx - this.camX, gcy - this.camY, 3, 0, Math.PI * 2);
          ctx.fill();
        }
        continue;
      }
      // Single door (no group) - original rendering
      const sx = door.tileX * ts - this.camX;
      const sy = door.tileY * ts - this.camY;
      if (sx + ts < -50 || sx > this.canvas.width + 50 || sy + ts < -50 || sy > this.canvas.height + 50) continue;
      const dx = door.tileX * ts + ts / 2;
      const dy = door.tileY * ts + ts / 2;
      let near = false;
      for (let pi = 0; pi < this.players.length; pi++) {
        const pl = this.players[pi];
        if (!pl.alive) continue;
        const pdx = pl.px - dx;
        const pdy = pl.py - dy;
        if (Math.sqrt(pdx * pdx + pdy * pdy) < doorRange) { near = true; break; }
      }
      if (!near) continue;
      ctx.fillStyle = INTERIOR_COLOR;
      ctx.fillRect(sx, sy, ts, ts);
      if (door.animating) {
        ctx.fillStyle = DOOR_CLOSED_COLOR;
        ctx.fillRect(sx + ts * 0.1, sy + ts * 0.1, ts * 0.8 * door.animProgress, ts * 0.8);
      } else if (!door.open) {
        ctx.fillStyle = DOOR_CLOSED_COLOR;
        ctx.fillRect(sx + ts * 0.1, sy + ts * 0.1, ts * 0.8, ts * 0.8);
      }
    }
  }

  renderNight(ctx) {
    if (this.nightTransition <= 0) return;
    // Vignette
    const cx = this.canvas.width / 2;
    const cy = this.canvas.height / 2;
    const maxR = Math.sqrt(cx * cx + cy * cy);
    const gradient = ctx.createRadialGradient(cx, cy, maxR * 0.3, cx, cy, maxR);
    gradient.addColorStop(0, 'rgba(0,0,0,0)');
    gradient.addColorStop(0.5, `rgba(0,0,30,${this.nightTransition * 0.3})`);
    gradient.addColorStop(1, `rgba(0,0,0,${this.nightTransition * 0.7})`);
    ctx.fillStyle = gradient;
    ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
  }

  renderDebug(ctx) {
    ctx.font = '10px monospace';
    ctx.fillStyle = '#0f0';
    // Waypoints
    for (const wp of this.waypoints) {
      const sx = wp.x * TILE_SIZE + TILE_SIZE / 2 - this.camX;
      const sy = wp.y * TILE_SIZE + TILE_SIZE / 2 - this.camY;
      ctx.strokeStyle = '#ff0';
      ctx.lineWidth = 1;
      ctx.strokeRect(sx - 4, sy - 4, 8, 8);
      ctx.fillText('U', sx + 6, sy - 6);
    }
    // Monster states
    for (const m of this.activeMonsters) {
      if (m.alive) {
        const mx = m.px - this.camX;
        const my = m.py - this.camY;
        ctx.fillStyle = '#f0f';
        ctx.fillText(m.type.toUpperCase() + ' ' + m.state.toUpperCase(), mx + 12, my - 12);
        m.renderVisionCone(ctx, this);
      }
    }
    // FPS
    ctx.fillText('FPS: ' + Math.round(1000 / (performance.now() - this.lastTime + 0.001)), 10, this.canvas.height - 10);
    // Player info
    let yOff = this.canvas.height - 45;
    for (const p of this.players) {
      const vr = this.playerVisionRange[p.playerIndex] || 0;
      ctx.fillText(p.name + ' Tile: ' + p.tileX + ',' + p.tileY + ' HP:' + p.hp + ' Class:' + p.playerClass + ' VisRng:' + (vr | 0), 10, yOff);
      yOff -= 15;
    }
    // Raycast perf info
    const totalRays = this.visibilityPolygons.reduce((s, vp) => s + vp.points.length, 0);
    ctx.fillText('Rays: ' + totalRays + ' | Diff: ' + this.difficulty + ' | Rnd: ' + this.roundNumber + ' | Spd:+' + this.getSpeedBonusPercent() + '%', 10, this.canvas.height - 30);
    ctx.fillText('Traps: ' + this.traps.length + ' | Shops: ' + this.shopItems.filter(s => !s.sold).length + ' | Cash: ' + this.currency, 10, this.canvas.height - 15);
    // Random spawn points
    for (const rp of this.randomMonsterSpawnPoints) {
      const sx = rp.x * TILE_SIZE + TILE_SIZE / 2 - this.camX;
      const sy = rp.y * TILE_SIZE + TILE_SIZE / 2 - this.camY;
      ctx.strokeStyle = '#0f0';
      ctx.lineWidth = 1;
      ctx.strokeRect(sx - 6, sy - 6, 12, 12);
      ctx.fillStyle = '#0f0';
      ctx.fillText('m', sx + 8, sy + 4);
    }
    // Monster type info
    const mt = MonsterTypes[this.monsterSpawnType];
    if (mt) {
      ctx.fillText('Monster: ' + mt.name + ' | SpdMod: ' + (mt.speedMod || 1).toFixed(1) + 'x | RndSpawn: ' + mt.randomSpawn, 10, this.canvas.height - 60);
    }
  }

  renderMessage(ctx) {
    const el = document.getElementById('message');
    if (this.messageText) {
      el.classList.add('show');
      el.textContent = this.messageText;
    } else {
      el.classList.remove('show');
    }
  }

  renderInteractionBar() {
    const label = document.getElementById('interaction-label');
    // Always hide the horizontal bar - replaced by canvas circle
    document.getElementById('interaction-bar').style.display = 'none';
    // Check if any player has interactTarget
    let hasTarget = false;
    let promptType = '';
    for (const p of this.players) {
      if (p.interactTarget && !p.interacting) {
        hasTarget = true;
        promptType = p.interactType;
        break;
      }
    }
    if (hasTarget) {
      label.style.display = 'block';
      if (promptType === 'door') label.textContent = 'Hold key to open/close';
      else if (promptType === 'altar') label.textContent = 'Hold key to summon';
      else if (promptType === 'artifact') label.textContent = 'Hold key to collect';
    } else {
      // Check if interacting to show active label
      let interactingPlayer = null;
      for (const p of this.players) {
        if (p.interacting && p.interactTarget) { interactingPlayer = p; break; }
      }
      if (interactingPlayer && this.interactDisplay) {
        label.style.display = 'block';
        if (this.interactDisplay.type === 'door') label.textContent = 'Opening/Closing...';
        else if (this.interactDisplay.type === 'altar') label.textContent = 'Summoning...';
        else if (this.interactDisplay.type === 'artifact') label.textContent = 'Collecting...';
      } else {
        label.style.display = 'none';
      }
    }
  }

  renderInteractionCircle(ctx) {
    if (!this.interactDisplay) return;
    const data = this.interactDisplay;
    const type = data.type;
    if (type !== 'artifact' && type !== 'altar') return;
    if (this.interactDuration <= 0) return;
    let interactingPlayer = null;
    for (const p of this.players) {
      if (p.interacting && p.interactTarget && p.interactType === type) {
        interactingPlayer = p; break;
      }
    }
    if (!interactingPlayer) return;
    const target = data.target;
    let worldX, worldY;
    if (type === 'artifact') {
      worldX = target.tileX * TILE_SIZE + TILE_SIZE / 2;
      worldY = target.tileY * TILE_SIZE + TILE_SIZE / 2;
    } else if (type === 'altar') {
      worldX = this.altarPos.x * TILE_SIZE + TILE_SIZE / 2;
      worldY = this.altarPos.y * TILE_SIZE + TILE_SIZE / 2;
    } else {
      worldX = target.x * TILE_SIZE + TILE_SIZE / 2;
      worldY = target.y * TILE_SIZE + TILE_SIZE / 2;
    }
    const sx = worldX - this.camX;
    const sy = worldY - this.camY;
    if (sx < -50 || sx > this.canvas.width + 50 || sy < -50 || sy > this.canvas.height + 50) return;
    const circleRadius = TILE_SIZE * 0.4;
    const lineWidth = circleRadius * 2 * 0.15;
    const progress = Math.min(1, Math.max(0, interactingPlayer.interactTimer / this.interactDuration));
    const startAngle = -Math.PI / 2;
    const endAngle = startAngle + Math.PI * 2 * progress;
    ctx.beginPath();
    ctx.arc(sx, sy, circleRadius, 0, Math.PI * 2);
    ctx.strokeStyle = '#333';
    ctx.lineWidth = lineWidth + 2;
    ctx.stroke();
    ctx.beginPath();
    ctx.arc(sx, sy, circleRadius, 0, Math.PI * 2);
    ctx.strokeStyle = '#555';
    ctx.lineWidth = lineWidth;
    ctx.stroke();
    if (progress > 0) {
      ctx.beginPath();
      ctx.arc(sx, sy, circleRadius, startAngle, endAngle);
      ctx.strokeStyle = Game.progressColor(progress);
      ctx.lineWidth = lineWidth;
      ctx.lineCap = 'round';
      ctx.stroke();
      ctx.lineCap = 'butt';
    }
  }
}

const keys = {};
const keysJustPressed = {};
window.addEventListener('keydown', e => {
  const k = e.key.toLowerCase();
  // Hotkey rebinding check
  if (rebindingPlayer !== null && rebindingAction !== null) {
    e.preventDefault();
    if (k === 'escape') {
      cancelRebinding();
      return;
    }
    finalizeRebinding(k);
    return;
  }
  if (!keys[k]) keysJustPressed[k] = true;
  keys[k] = true;
  if (e.key === 'F1') {
    e.preventDefault();
    game.debugMode = !game.debugMode;
  }
  if (game && game.players) {
    for (const p of game.players) {
      if (!p.alive) continue;
      if (k === p.keybinds.ability) {
        activateAbility(game, p);
      }
      if (k === p.keybinds.interact && !p.interacting && p.interactTarget) {
        if (p.interactType !== 'door') {
          p.interacting = true;
          p.interactTimer = 0;
        }
      }
    }
  }
});
window.addEventListener('keyup', e => {
  const k = e.key.toLowerCase();
  keys[k] = false;
  if (game && game.players) {
    for (const p of game.players) {
      if (k === p.keybinds.interact) {
        p.interacting = false;
        p.interactTimer = 0;
      }
    }
  }
});

function activateAbility(game, player) {
  const p = player;
  const cls = PlayerClasses[p.playerClass];
  if (!cls || cls.cooldown === 0) return;
  if (p.abilityCooldown > 0 || p.abilityActive) return;
  p.abilityActive = true;
  p.abilityTimer = cls.duration;
  p.abilityCooldown = cls.cooldown + cls.duration;
  if (p.playerClass === 'scout') {
    p.speed = p.baseSpeed * 1.5;
  }
}

function updateHUD(game) {
  const hud = document.getElementById('hud');
  const night = game.nightTransition > 0.3 ? 'NIGHT' : 'DAY';
  const mt = game.monsterSpawnType;
  const mtName = MonsterTypes[mt] ? MonsterTypes[mt].name : 'None';
  let html = '';
  for (const p of game.players) {
    const clsName = PlayerClasses[p.playerClass] ? PlayerClasses[p.playerClass].name : 'None';
    html += '<span style="color:' + (['#4488ff','#ff8844','#44ff44','#ff44ff'][p.playerIndex] || '#fff') + '">' + p.name + '</span>: ';
    html += 'HP:' + p.hp + '/' + p.maxHp + ' ';
    html += clsName + ' ';
    html += (p.alive ? '' : 'DEAD ') + '<br>';
  }
  html += 'Monster: ' + mtName + '<br>';
  html += 'Difficulty: ' + game.difficulty + ' | Round: ' + game.roundNumber + '<br>';
  html += 'Artefatos: ' + game.collectedArtifacts + '/' + game.artifactCount + '<br>';
  html += 'Speed Bonus: +' + game.getSpeedBonusPercent() + '%<br>';
  html += 'Coins: ' + game.currency + '<br>';
  html += 'Estado: ' + night;
  hud.innerHTML = html;
}

function updateAbilityHUD(game) {
  const hud = document.getElementById('ability-hud');
  let html = '';
  let anyAbility = false;
  for (const p of game.players) {
    if (!p.alive) continue;
    const cls = PlayerClasses[p.playerClass];
    if (!cls || cls.cooldown === 0) continue;
    anyAbility = true;
    const color = ['#4488ff','#ff8844','#44ff44','#ff44ff'][p.playerIndex] || '#ccc';
    html += '<div style="text-align:left; margin-bottom:4px;">';
    html += '<span style="color:' + color + '">' + p.name + '</span> ';
    html += cls.abilityName + ' [' + p.keybinds.ability.toUpperCase() + '] ';
    if (p.abilityActive) {
      const pct = (p.abilityTimer / cls.duration) * 100;
      html += '<span style="color:#ff0">Active ' + p.abilityTimer.toFixed(1) + 's</span>';
    } else if (p.abilityCooldown > 0) {
      html += '<span style="color:#f66">CD ' + p.abilityCooldown.toFixed(1) + 's</span>';
    } else {
      html += '<span style="color:#6f6">Ready</span>';
    }
    html += '</div>';
  }
  if (!anyAbility) {
    hud.style.display = 'none';
    return;
  }
  hud.style.display = 'block';
  hud.innerHTML = html;
}

function updateHotbarHUD(game) {
  const container = document.getElementById('hotbar-hud');
  let html = '';
  for (const p of game.players) {
    if (!p.alive) continue;
    const color = ['#4488ff','#ff8844','#44ff44','#ff44ff'][p.playerIndex] || '#ccc';
    html += '<div style="display:flex;gap:8px;align-items:center;margin-right:30px;">';
    html += '<span style="color:' + color + ';font-size:23px;min-width:50px;">' + p.name.substring(0, 3) + '</span>';
    for (let i = 0; i < 5; i++) {
      const item = p.inventory[i];
      const selected = i === p.selectedSlot;
      const it = item ? Items[item] : null;
      html += '<div style="width:55px;height:55px;background:' + (selected ? '#2a2a3e' : '#111') + ';border:1px solid ' + (selected ? '#4fc3f7' : '#333') + ';border-radius:8px;display:flex;align-items:center;justify-content:center;font-size:28px;color:' + (it ? it.color : '#333') + ';">';
      html += it ? it.icon : '-';
      html += '</div>';
    }
    if (p.useTimer > 0) {
      const pct = Math.min(100, (p.useTimer / 1) * 100);
      html += '<div style="width:50px;height:10px;background:rgba(255,255,255,0.1);border-radius:5px;margin-left:10px;">';
      html += '<div style="height:100%;width:' + pct + '%;background:#4fc3f7;border-radius:5px;transition:width 0.05s;"></div>';
      html += '</div>';
    }
    html += '</div>';
  }
  container.innerHTML = html;
}

function updateMapEditor(game) {
  // Build legend
  const legend = document.getElementById('map-legend');
  let html = '';
  for (const [ch, label] of Object.entries(LEGEND_LABELS)) {
    const color = LEGEND_COLORS[ch];
    html += '<span style="color:' + color + '">' + ch + ' = ' + label + '</span>';
  }
  legend.innerHTML = html;
}

function updatePreview() {
  const ta = document.getElementById('map-editor-textarea');
  const pre = document.querySelector('#map-editor-preview pre');
  const text = ta.value;
  let html = '';
  for (const ch of text) {
    if (ch === '\n') { html += '\n'; continue; }
    if (ch === ' ') { html += ' '; continue; }
    const color = LEGEND_COLORS[ch] || '#ccc';
    html += '<span style="color:' + color + '">' + ch + '</span>';
  }
  pre.innerHTML = html;
}

let rebindingPlayer = null;
let rebindingAction = null;

function buildHotkeyConfig(game) {
  const container = document.getElementById('hotkey-config');
  let html = '<h3>Key Bindings</h3>';
  for (let i = 0; i < game.playerCount; i++) {
    const p = game.players[i];
    if (!p) continue;
    const color = ['#4488ff','#ff8844','#44ff44','#ff44ff'][i] || '#ccc';
    html += '<div style="margin-bottom:8px;">';
    html += '<h3 style="color:' + color + '">Player ' + (i + 1) + ' (' + p.name + ')</h3>';
    const actions = [
      { key: 'up', label: 'Move Up' },
      { key: 'down', label: 'Move Down' },
      { key: 'left', label: 'Move Left' },
      { key: 'right', label: 'Move Right' },
      { key: 'interact', label: 'Interact' },
      { key: 'ability', label: 'Ability' },
      { key: 'use', label: 'Use/Cycle Item' }
    ];
    for (const act of actions) {
      html += '<div class="hotkey-row">';
      html += '<span class="action">' + act.label + '</span>';
      html += '<button class="key-btn" data-player="' + i + '" data-action="' + act.key + '">' + p.keybinds[act.key].toUpperCase() + '</button>';
      html += '</div>';
    }
    html += '</div>';
  }
  container.innerHTML = html;
  // Add event listeners
  container.querySelectorAll('.key-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const pi = parseInt(btn.dataset.player);
      const act = btn.dataset.action;
      rebindingPlayer = pi;
      rebindingAction = act;
      container.querySelectorAll('.key-btn').forEach(b => b.classList.remove('waiting'));
      btn.classList.add('waiting');
      btn.textContent = '...';
    });
  });
}

function finalizeRebinding(keyName) {
  if (rebindingPlayer === null || rebindingAction === null) return false;
  const p = game.players[rebindingPlayer];
  if (!p) { rebindingPlayer = null; rebindingAction = null; return false; }
  // Clear old binding from keys
  keys[p.keybinds[rebindingAction]] = false;
  p.keybinds[rebindingAction] = keyName;
  rebindingPlayer = null;
  rebindingAction = null;
  buildHotkeyConfig(game);
  return true;
}

function cancelRebinding() {
  rebindingPlayer = null;
  rebindingAction = null;
  buildHotkeyConfig(game);
}

    game.restart();
    document.getElementById('settings-panel').classList.remove('open');
  });

  document.getElementById('btn-map-editor').addEventListener('click', () => {
    game.openMapEditor();
  });

  document.getElementById('btn-apply-map').addEventListener('click', () => {
    game.applyMapFromEditor();
  });

  document.getElementById('btn-reset-map').addEventListener('click', () => {
    game.resetDefaultMap();
  });

  document.getElementById('btn-cancel-map').addEventListener('click', () => {
    game.closeMapEditor();
  });

  // Live preview update
  document.getElementById('map-editor-textarea').addEventListener('input', updatePreview);

  // Close panels on Escape
  window.addEventListener('keydown', e => {
    if (e.key === 'Escape') {
      document.getElementById('settings-panel').classList.remove('open');
      game.closeMapEditor();
      cancelRebinding();
    }
  });

  // Resize canvas display
  function resizeCanvas() {
    const container = document.getElementById('game-container');
    const maxW = container.clientWidth - 40;
    const maxH = container.clientHeight - 40;
    const canvas = document.getElementById('game');
    const scaleW = maxW / canvas.width;
    const scaleH = maxH / canvas.height;
    const scale = Math.min(scaleW, scaleH, 1.5);
    canvas.style.width = (canvas.width * scale) + 'px';
    canvas.style.height = (canvas.height * scale) + 'px';
  }
  window.addEventListener('resize', resizeCanvas);
  resizeCanvas();
});
